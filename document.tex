	\chapter{مقدمه}
	\vspace{-2.5\baselineskip}
	\noindent\rule{\textwidth}{0.4pt}
	
	
	\hspace{-0.1cm} 
	با گسترش روزافزون اینترنت و افزایش نیاز به خدمات آنلاین، پلتفرم‌های واسطی که ارتباط بین ارائه‌دهندگان خدمات و مشتریان را تسهیل می‌کنند، اهمیت ویژه‌ای پیدا کرده‌اند همچنین در بسیاری از حوزه‌ها، از جمله مدیریت خوابگاه‌های دانشجویی، یافتن یک سیستم مدیریتی قابل اعتماد و کارآمد برای رسیدگی به نیازهای ساکنان می‌تواند چالش‌برانگیز باشد. از سوی دیگر، مدیران خوابگاه‌ها نیز نیاز به یک بستر کارآمد برای مدیریت ثبت‌نام‌ها، نظارت بر درخواست‌های تعمیرات، و برقراری ارتباط مؤثر با دانشجویان دارند.
	
	پروژه «سیستم مدیریت خوابگاه» با هدف پاسخ به این نیازها، به عنوان یک پلتفرم آنلاین طراحی و پیاده‌سازی شده است که فرآیند مدیریت خوابگاه، پیگیری درخواست‌ها، و ارتباط بین اعضای مختلف را تسهیل می‌کند. این سیستم به سه دسته اصلی از کاربران خدمت‌رسانی می‌کند: \textbf{دانشجویان} که نیاز به ثبت‌نام در خوابگاه و دسترسی به خدمات مختلف دارند، \textbf{متخصصان سرویس} که در حوزه‌های مختلف تعمیرات و نگهداری تخصص دارند و به دنبال دریافت و انجام درخواست‌های کاری می‌گردند، و \textbf{مدیران} که مسئولیت نظارت، تایید، و کنترل کلیه فرآیندها را بر عهده دارند.
	
	هدف اصلی این پروژه، طراحی و پیاده‌سازی یک پلتفرم جامع برای مدیریت هوشمند خوابگاه‌های دانشجویی در یک محیط دیجیتالی امن و کارآمد است. این اهداف به شرح زیر هستند:
	
	\begin{itemize}
		\item \textbf{تسهیل ارتباط}: ایجاد یک پل ارتباطی مستقیم و کارآمد بین دانشجویان، متخصصان سرویس، و مدیریت خوابگاه.
		
		\item \textbf{شفافیت و امنیت}: فراهم کردن یک سیستم احراز هویت قوی، تایید ایمیل، و مکانیزم‌های نظارتی برای افزایش اعتماد در پلتفرم.
		
		\item \textbf{مدیریت هوشمند درخواست‌ها}: ایجاد سیستمی برای ثبت درخواست‌های تعمیرات با امکان بارگذاری تصاویر، تعیین اولویت توسط مدیران، تخصیص به متخصصان مربوطه، و پیگیری وضعیت.
		
		\item \textbf{مقیاس‌پذیری}: طراحی معماری به گونه‌ای که امکان گسترش سیستم برای افزودن قابلیت‌های بیشتر و پشتیبانی از تعداد بیشتری از کاربران در آینده فراهم باشد.
		
		
		\item \textbf{کنترل کیفیت}: پیاده‌سازی سیستم امتیازدهی و بازخورد برای ارزیابی کیفیت خدمات ارائه شده توسط متخصصان سرویس.
	\end{itemize}
	
	\section{انگیزه و زمینه پروژه}
	\noindent\rule{\textwidth}{0.4pt}
	
	\hspace{-0.1cm}
	ایده طراحی و پیاده‌سازی این سیستم از مشاهده مستقیم چالش‌های روزمره دانشجویان ساکن در خوابگاه‌های دانشگاهی نشأت گرفته است. در بسیاری از خوابگاه‌ها، فرآیند ثبت و پیگیری درخواست‌های تعمیرات به صورت دستی و از طریق مراجعه حضوری به دفتر مدیریت انجام می‌پذیرد که این امر نه تنها زمان‌بر و ناکارآمد است، بلکه امکان پیگیری شفاف و مستندسازی دقیق را نیز فراهم نمی‌آورد.
	
	مشاهدات میدانی نشان داد که دانشجویان گاهاً مجبور به انتظار چند روزه برای رسیدگی به مشکلات ساده بوده و هیچ اطلاعی از وضعیت درخواست خود نداشتند. از سوی دیگر، مدیران خوابگاه نیز با حجم بالای درخواست‌های شفاهی و کاغذی دست و پنجه نرم کرده و امکان اولویت‌بندی علمی و ارزیابی عملکرد متخصصان را نداشتند. این شرایط انگیزه اصلی برای طراحی یک سامانه جامع و دیجیتال را فراهم آورد.
	
	\section{دامنه و محدوده پروژه}

	\subsection{موارد تحت پوشش}
	
	این سیستم در نسخه فعلی خود، بخش‌های زیر را پوشش می‌دهد:
	
	\begin{itemize}
		\item \textbf{مدیریت کاربران}: ثبت‌نام، احراز هویت و مدیریت سه نوع کاربر (دانشجو، متخصص، مدیر)
		\item \textbf{مدیریت درخواست‌های تعمیرات}: ثبت، اولویت‌بندی، تخصیص، پیگیری و تکمیل
		\item \textbf{سیستم ارتباطات}: چت آنلاین و تیکتینگ برای ارتباط مستقیم
		\item \textbf{ارزیابی خدمات}: امتیازدهی و بازخورد پس از تکمیل کار
		\item \textbf{مدیریت اطلاعات اولیه}: ساختمان‌ها و اتاق‌های خوابگاه
	\end{itemize}
	
	\subsection{موارد خارج از محدوده (نسخه فعلی)}
	
	موارد زیر در نسخه فعلی سیستم پیاده‌سازی نشده و برای توسعه‌های آینده در نظر گرفته شده‌اند:
	
	\begin{itemize}
		\item مدیریت مالی و حسابداری خوابگاه
		\item سیستم رزرو امکانات مشترک (سالن ورزشی، اتاق مطالعه)
		\item مدیریت ورود و خروج روزانه ساکنین
		\item سیستم اطلاع‌رسانی عمومی و انتشار اخبار
		\item پنل تحلیلی و گزارش‌های پیشرفته مدیریتی
	\end{itemize}
	
	
	
	\section{نوآوری‌های سیستم}
	\noindent\rule{\textwidth}{0.4pt}
	
	این سیستم دارای ویژگی‌های منحصر به فردی است که آن را از سایر سیستم‌های مشابه متمایز می‌کند:
	
	\begin{itemize}
		\item \textbf{احراز هویت چندگانه}: پشتیبانی از سه نوع ورود مختلف (شماره دانشجویی، کد کارمندی، ایمیل مدیر) با مکانیزم‌های امنیتی جداگانه
		
		\item \textbf{تأیید ایمیل زمان‌دار}: سیستم خودکار حذف حساب‌های تأیید نشده پس از ۱۵ دقیقه برای افزایش امنیت و جلوگیری از ثبت‌نام‌های جعلی
		
		\item \textbf{مستندسازی تصویری الزامی}: الزام به آپلود تصویر مشکل در هنگام ثبت درخواست و تصویر نتیجه کار پس از تکمیل
		
		\item \textbf{ارتباطات \lr{Real-time}}: استفاده از \lr{WebSocket} برای چت آنی و به‌روزرسانی لحظه‌ای بدون نیاز به رفرش صفحه
		
		\item \textbf{سیستم اولویت‌بندی هوشمند}: مدیران می‌توانند اولویت درخواست‌ها را تعیین کنند و سیستم جلوی تأیید درخواست‌های بدون اولویت را می‌گیرد
		
		\item \textbf{ارزیابی دوطرفه}: هم دانشجویان می‌توانند متخصصان را امتیازدهی کنند و هم امتیازات در تصمیمات آتی مؤثر است
	\end{itemize}
	
	
	
	
	\section{ساختار مستند}
	\noindent\rule{\textwidth}{0.4pt}
	
	این مستند در پنج فصل اصلی سازماندهی شده است:
	
	\textbf{فصل اول - مقدمه}: شامل معرفی کلی پروژه، اهداف، انگیزه‌ها، دامنه کاری و نمودارهای طراحی اولیه سیستم می‌باشد.
	
	\textbf{فصل دوم - پیشینه تحقیق}: در این فصل، سیستم‌های موجود مدیریت خوابگاه در سطح داخلی و بین‌المللی بررسی شده و شکاف‌های موجود شناسایی می‌گردد تا ضرورت توسعه سیستم جدید مستند شود.
	
	\textbf{فصل سوم - طریقه کار با سیستم}: راهنمای جامع استفاده از سیستم برای هر یک از سه نوع کاربر با تصاویر و مثال‌های عملی ارائه می‌شود.
	
	\textbf{فصل چهارم - تکنولوژی‌های مورد استفاده}: معماری فنی سیستم، تکنولوژی‌های به کار رفته در بخش‌های مختلف و دلایل انتخاب هر یک تشریح می‌گردد.
	
	\textbf{فصل پنجم - نتیجه‌گیری و پیشنهادات}: دستاوردهای پروژه، چالش‌های پیاده‌سازی، درس‌های آموخته شده و پیشنهادات برای توسعه‌های آینده مطرح می‌شود.

	\subsection{مدیریت کاغذی}
	
	در بسیاری از مراکز آموزشی و دانشگاه‌های کشور، مدیریت خوابگاه همچنان به صورت سنتی و با استفاده از اسناد کاغذی انجام می‌شود. این روش شامل فرآیندهای دستی ثبت‌نام، تخصیص اتاق، و مدیریت درخواست‌های تعمیرات است.
	
	\subsubsection*{ویژگی‌های سیستم کاغذی}
	\begin{itemize}
		\item استفاده از فرم‌های کاغذی برای ثبت‌نام و درخواست‌های مختلف
		\item نگهداری پرونده‌های فیزیکی دانشجویان
		\item استفاده از دفاتر ثبت ورود و خروج
		\item مدیریت دستی درخواست‌های تعمیرات
	\end{itemize}
	
	\subsubsection*{مزایای روش سنتی}
	\begin{itemize}
		\item عدم نیاز به زیرساخت تکنولوژیکی پیشرفته
		\item آشنایی کارکنان با فرآیندهای موجود
		\item هزینه اولیه پایین برای پیاده‌سازی
		\item امکان دسترسی بدون نیاز به تجهیزات الکترونیکی
	\end{itemize}
	
	\subsubsection*{محدودیت‌های عمده}
	\begin{itemize}
		\item زمان‌بر بودن فرآیندها و نیاز به حضور فیزیکی
		\item امکان گم شدن یا تخریب اسناد و مدارک
		\item عدم شفافیت در وضعیت درخواست‌ها و عدم امکان پیگیری آنلاین
		\item محدودیت در گزارش‌گیری دقیق و تحلیل آمارها
		\item عدم امکان اطلاع‌رسانی سریع به دانشجویان
		\item نیاز به نیروی انسانی بیشتر برای مدیریت فرآیندها
	\end{itemize}
	
	\subsection{سیستم‌های نیمه‌دیجیتال}

	\subsubsection*{ابزارهای متداول}
	\begin{itemize}

		\item نرم‌افزارهای پایگاه داده محلی مانند Access
		\item سیستم‌های مدیریت محتوای ساده
	\end{itemize}
	
	\subsubsection*{نقاط ضعف رویکرد نیمه‌دیجیتال}
	\begin{itemize}
		\item عدم یکپارچگی اطلاعات در بخش‌های مختلف
		\item محدودیت در دسترسی همزمان چندین کاربر
		\item نبود امکانات امنیتی مناسب

	سیستم‌های داخلی نیز عمدتاً بر جنبه‌های مدیریتی پایه تمرکز داشته و فاقد قابلیت‌های پیشرفته ارتباطی، مدیریت تعمیرات و کنترل کیفیت هستند.
	
	\hspace{0.000001cm}  سیستم پیشنهادی با ترکیب نوآوری‌های فناورانه شامل سیستم چت بلادرنگ، مدیریت تصویری درخواست‌ها، سیستم رتبه‌دهی جامع و مدیریت هوشمند اولویت‌ها، قادر است شکاف‌های موجود را پر کرده و راه‌حلی جامع و کارآمد برای مدیریت خوابگاه‌های دانشجویی ارائه دهد.
	
	این سیستم نه تنها نیازهای فعلی را برطرف می‌کند، بلکه با طراحی مقیاس‌پذیر، امکان توسعه و افزودن قابلیت‌های جدید در آینده را نیز فراهم می‌آورد
	
	
	
	\chapter{تحلیل و طراحی سیستم}
	\vspace{-2.5\baselineskip}
	\noindent\rule{\textwidth}{0.4pt}
	
	\hspace{-0.1cm}
	تحلیل و طراحی سیستم یکی از مهم‌ترین و حیاتی‌ترین مراحل در فرآیند توسعه نرم‌افزار محسوب می‌شود که پایه و اساس موفقیت پروژه را تشکیل می‌دهد. در این مرحله، نیازمندی‌های سیستم به دقت شناسایی، تحلیل و به مدل‌های قابل فهم و قابل پیاده‌سازی تبدیل می‌شوند. یک تحلیل و طراحی دقیق و جامع، نه تنها مسیر توسعه را روشن می‌کند، بلکه از بروز مشکلات و هزینه‌های اضافی در مراحل بعدی جلوگیری می‌نماید.
	
	در پروژه سیستم مدیریت خوابگاه، با توجه به تنوع کاربران (دانشجویان، متخصصان و مدیران)، پیچیدگی فرآیندهای کسب‌وکار و نیاز به تعاملات \lr{Real-time}، تحلیل و طراحی دقیق سیستم از اهمیت بالایی برخوردار است. در این فصل، به بررسی جامع ساختار سیستم، روابط بین اجزای مختلف و تعاملات کاربران با سیستم پرداخته می‌شود.
	
	\section{اهداف تحلیل و طراحی}
	
	\noindent\rule{\textwidth}{0.4pt}
	
	اهداف اصلی این فصل عبارتند از:
	
	\begin{itemize}
		\item \textbf{مدل‌سازی ساختار داده}: ارائه تصویری جامع از کلاس‌های اصلی سیستم و روابط بین آنها از طریق نمودار کلاس که نحوه سازماندهی اطلاعات و منطق کسب‌وکار را مشخص می‌کند
		
		\item \textbf{تعریف تعاملات کاربر-سیستم}: شناسایی و مستندسازی تمامی موارد استفاده (\lr{Use Cases}) که نشان می‌دهد هر نوع کاربر چه عملیاتی را می‌تواند انجام دهد و سیستم چگونه به درخواست‌های آنها پاسخ می‌دهد
		
		\item \textbf{تضمین پوشش کامل نیازمندی‌ها}: اطمینان از اینکه تمامی نیازمندی‌های کاربران در طراحی سیستم لحاظ شده و هیچ قابلیت حیاتی فراموش نشده است
		
		\item \textbf{ایجاد راهنمای پیاده‌سازی}: فراهم کردن مستندات واضح و دقیق برای تیم توسعه جهت پیاده‌سازی صحیح سیستم
		
		\item \textbf{تسهیل ارتباط با ذینفعان}: ارائه مدل‌های بصری و قابل فهم که امکان بررسی و تأیید طراحی توسط ذینفعان مختلف پروژه را فراهم می‌آورد
	\end{itemize}
	
	\section{روش‌شناسی طراحی}
	
	\noindent\rule{\textwidth}{0.4pt}
	
	در این پروژه از رویکرد شی‌گرا (\lr{Object-Oriented}) برای تحلیل و طراحی سیستم استفاده شده است. این رویکرد مزایای متعددی دارد:
	
	\subsection{متدولوژی توسعه}
	
	برای مدیریت فرآیند توسعه این سیستم، از متدولوژی \textbf{آبشاری (\lr{Waterfall})} استفاده شده است. با توجه به ماهیت تک‌نفره این پروژه و مشخص بودن نیازمندی‌های اولیه سیستم، این رویکرد انتخاب مناسبی محسوب می‌شود.
	
	\textbf{دلایل انتخاب متدولوژی آبشاری:}
	
	\begin{itemize}
		\item \textbf{پروژه تک‌نفره}: با توجه به اینکه تمامی مراحل توسعه توسط یک نفر انجام می‌شود، نیاز به هماهنگی تیمی و جلسات اسکرام وجود ندارد و مدل آبشاری با مراحل خطی و مشخص، مدیریت پروژه را ساده‌تر می‌کند
		
		\item \textbf{نیازمندی‌های مشخص}: نیازمندی‌های سیستم مدیریت خوابگاه از ابتدا به صورت کامل شناسایی و مستند شده‌اند که این امر امکان طراحی کامل سیستم قبل از شروع پیاده‌سازی را فراهم می‌آورد
		
		\item \textbf{ساختار منظم}: مراحل مشخص شامل تحلیل نیازمندی‌ها، طراحی سیستم، پیاده‌سازی، تست و استقرار به صورت متوالی و منظم دنبال شده‌اند
		
		\item \textbf{مستندسازی جامع}: در هر مرحله مستندات کامل تهیه شده که برای نگهداری و توسعه‌های آینده سیستم ضروری است
	\end{itemize}
	
	\textbf{مراحل اجرایی پروژه:}
	
	\begin{enumerate}
		\item \textbf{تحلیل نیازمندی‌ها}: شناسایی و مستندسازی نیازهای دانشجویان، متخصصان و مدیران
		\item \textbf{طراحی سیستم}: ایجاد نمودارهای \lr{UML}، طراحی پایگاه داده و معماری کلی
		\item \textbf{پیاده‌سازی}: کدنویسی بخش‌های مختلف سیستم با رعایت اصول طراحی
		\item \textbf{تست}: آزمایش عملکرد، امنیت و قابلیت استفاده سیستم
		\item \textbf{استقرار}: نصب و راه‌اندازی سیستم در محیط تولید
		\item \textbf{نگهداری}: رفع باگ‌ها و اضافه کردن قابلیت‌های جدید
	\end{enumerate}
	
	\subsection{مزایای رویکرد شی‌گرا}
	
	\begin{itemize}
		\item \textbf{قابلیت بازاستفاده}: کلاس‌ها و اجزای طراحی شده می‌توانند در بخش‌های مختلف سیستم یا حتی پروژه‌های آینده مورد استفاده قرار گیرند
		
		\item \textbf{انعطاف‌پذیری}: امکان تغییر و گسترش سیستم بدون تأثیر بر سایر بخش‌ها به دلیل کپسوله‌سازی و انتزاع مناسب
		
		\item \textbf{نگهداری آسان‌تر}: ساختار منطقی و سازمان‌یافته کد، فرآیند رفع اشکال و اضافه کردن قابلیت‌های جدید را ساده‌تر می‌کند
		
		\item \textbf{مدل‌سازی نزدیک به دنیای واقعی}: اشیاء و کلاس‌ها مفاهیم دنیای واقعی مانند دانشجو، متخصص، درخواست و اتاق را به خوبی مدل می‌کنند
	\end{itemize}
	\subsection{ابزارهای مدل‌سازی}
	
	برای مدل‌سازی و طراحی سیستم از زبان مدل‌سازی یکپارچه (\lr{UML - Unified Modeling Language}) استفاده شده که شامل:
	
	\begin{itemize}
		\item \textbf{نمودار کلاس (\lr{Class Diagram})}: برای نمایش ساختار استاتیک سیستم، کلاس‌ها، ویژگی‌ها، متدها و روابط بین آنها
		
		\item \textbf{نمودار مورد استفاده (\lr{Use Case Diagram})}: برای نمایش تعاملات بین کاربران (اکتورها) و سیستم، و قابلیت‌های ارائه شده
		
		\item \textbf{نمودار توالی (\lr{Sequence Diagram})}: برای نمایش جریان زمانی پیام‌ها و تعاملات بین اشیاء در سناریوهای مختلف (در صورت نیاز)
	\end{itemize}

	\subsection{تحلیل کلاس‌های اصلی}
	
	\subsubsection*{کلاس \texttt{User}}
	
	کلاس \texttt{User} کلاس پایه برای تمامی کاربران سیستم است و ویژگی‌های مشترک همه کاربران را در خود جای داده است. این کلاس شامل فیلدهای زیر است:
	
	\begin{itemize}
		\item \texttt{email}: آدرس ایمیل کاربر برای احراز هویت و ارتباطات
		\item \texttt{username}: نام کاربری منحصربه‌فرد
		\item \texttt{phone}: شماره تماس کاربر
		\item \texttt{user\_type}: نوع کاربر (دانشجو یا متخصص)
		\item \texttt{is\_active}: وضعیت فعالیت حساب کاربری
		\item \texttt{is\_staff}: دسترسی به پنل مدیریت
		\item \texttt{date\_joined}: تاریخ عضویت در سیستم
	\end{itemize}
	
	\subsubsection*{کلاس \texttt{Student}}
	
	این کلاس اطلاعات اختصاصی دانشجویان را نگهداری می‌کند:
	
	\begin{itemize}
		\item \texttt{student\_number}: شماره دانشجویی ۹ رقمی منحصربه‌فرد
		\item \texttt{room}: رابطه با کلاس \texttt{Room} برای تعیین محل اقامت
		\item \texttt{registration\_status}: وضعیت ثبت‌نام (در انتظار، تأیید شده، رد شده)
		\item \texttt{verification\_token}: توکن منحصربه‌فرد برای تأیید ایمیل
		\item رابطه \texttt{one-to-many} با \texttt{MaintenanceRequest} برای درخواست‌های ثبت شده
	\end{itemize}
	
	\subsubsection*{کلاس \texttt{ServiceExpert}}
	
	این کلاس اطلاعات متخصصان خدمات را مدیریت می‌کند:
	
	\begin{itemize}
		\item \texttt{employee\_id}: کد کارمندی منحصربه‌فرد
		\item \texttt{specialization}: تخصص متخصص (برق، تأسیسات، نجاری و...)
		\item \texttt{average\_rating}: میانگین امتیاز دریافتی از دانشجویان
		\item رابطه \texttt{one-to-many} با \texttt{MaintenanceRequest} برای درخواست‌های اختصاص یافته
	\end{itemize}
	
	\subsubsection*{کلاس \texttt{Building} و \texttt{Room}}
	
	کلاس‌های \texttt{Building} و \texttt{Room} ساختار فیزیکی خوابگاه را مدل‌سازی کرده‌اند:
	
	\begin{itemize}
		\item \texttt{Building}: شامل نام ساختمان، تعداد طبقات و تاریخ ایجاد
		\item \texttt{Room}: شامل شماره اتاق، طبقه، ظرفیت و رابطه با ساختمان مربوطه
		\item رابطه \texttt{one-to-many} بین \texttt{Building} و \texttt{Room}
	\end{itemize}
	
	\subsubsection*{کلاس \texttt{MaintenanceRequest}}
	
	این کلاس هسته اصلی سیستم است و فرآیند درخواست و انجام تعمیرات را مدیریت می‌کند:
	
	\begin{itemize}
		\item فیلدهای اطلاعاتی: عنوان، توضیحات، نوع خدمات، اولویت
		\item فیلدهای وضعیت: \texttt{status} (در انتظار، تأیید شده، در حال انجام، تکمیل شده)
		\item روابط: ارتباط با دانشجو، متخصص و اتاق
		\item فیلدهای زمانی: تاریخ ثبت، تأیید، شروع کار و تکمیل
		\item فیلدهای ارزیابی: امتیاز و بازخورد دانشجو
		\item رابطه \texttt{one-to-many} با \texttt{MaintenanceImage} برای تصاویر
	\end{itemize}
	
	\subsubsection*{کلاس \texttt{Ticket} و \texttt{TicketMessage}}
	
	سیستم تیکت‌ها از طریق این دو کلاس برای برقراری ارتباط در زمان واقعی بین کاربران پیاده‌سازی شده است:
	
	\begin{itemize}
		\item \texttt{Ticket}: شامل عنوان، توضیحات، وضعیت و ارتباط با ایجادکننده
		\item \texttt{TicketMessage}: شامل محتوای پیام، نویسنده، نوع پیام (ادمین یا کاربر) و امکان ضمیمه تصویر
		\item رابطه \texttt{one-to-many} بین \texttt{Ticket} و \texttt{TicketMessage}
	\end{itemize}
	
	\subsection{روابط بین کلاس‌ها}
	
	نمودار کلاس نشان می‌دهد که سیستم دارای روابط پیچیده و منطقی بین کلاس‌های مختلف است:
	
	\begin{itemize}
		\item رابطه \textbf{وراثت (\lr{Inheritance})}: بین \texttt{User} و کلاس‌های \texttt{Student}، \texttt{ServiceExpert}
		\item رابطه \textbf{یک‌به‌چند (\lr{One-to-Many})}: بین \texttt{Student} و \texttt{MaintenanceRequest}، بین \texttt{ServiceExpert} و \texttt{MaintenanceRequest}
		\item رابطه \textbf{کلید خارجی (\lr{Foreign Key})}: برای ایجاد ارتباط بین جداول در پایگاه داده
	\end{itemize}

	\section{نمودارهای فعالیت (\lr{Activity Diagrams})}
	
	\noindent\rule{\textwidth}{0.4pt}
	
	نمودارهای فعالیت جریان کنترل و توالی فعالیت‌ها در فرآیندهای کسب‌وکار را مدل‌سازی می‌کنند. این نمودارها جزئیات دقیق‌تری از مراحل اجرایی، تصمیم‌گیری‌ها و جریان داده‌ها ارائه می‌دهند که برای درک فرآیندهای پیچیده شامل چندین بازیگر ضروری است.
	
	\subsection{نمودار فعالیت فرآیند ثبت‌نام دانشجویان}
	
	فرآیند ثبت‌نام اولیه دانشجویان شامل مراحل اعتبارسنجی، تأیید ایمیل و بررسی نهایی توسط مدیر است که به عنوان دروازه ورودی به سیستم عمل می‌کند.
	
	\begin{figure}[H]
		\centering
		\resizebox{0.75\textwidth}{!}{%
			\includegraphics{../issue/activity/Register.png}%
		}
		\caption{نمودار فعالیت فرآیند ثبت‌نام دانشجویان}
		\label{fig:student_registration_activity}
	\end{figure}
	
	همان‌طور که در شکل \ref{fig:student_registration_activity} مشاهده می‌شود، فرآیند شامل سه شناگر اصلی (دانشجو، سیستم، مدیر) است:
	
	\textbf{مراحل اصلی فرآیند:}
	
	\begin{enumerate}
		\item \textbf{تکمیل فرم}: دانشجو اطلاعات شخصی و تحصیلی را وارد کرده و سیستم اعتبارسنجی‌های اولیه را انجام می‌دهد
		
		\item \textbf{ایجاد حساب غیرفعال}: سیستم حساب کاربری غیرفعال ایجاد کرده و لینک تأیید را از طریق ایمیل ارسال می‌کند
		
		\item \textbf{فعال‌سازی}: دانشجو با کلیک بر روی لینک، حساب را فعال کرده و وضعیت به \textbf{Pending} تغییر می‌کند
		
		\item \textbf{بررسی مدیر}: مدیر اطلاعات را بررسی کرده و تصمیم به تأیید یا رد می‌گیرد
		
		\item \textbf{نتیجه}: در صورت تأیید، ایمیل اطلاعات ورود ارسال شده و دانشجو می‌تواند وارد سیستم شود. در صورت رد، ایمیل حاوی دلیل رد ارسال می‌گردد
	\end{enumerate}
	
	این فرآیند دارای امنیت چندلایه (تأیید ایمیل + بررسی دستی)، جداسازی مسئولیت‌ها و قابلیت پیگیری کامل است.

	دانشجویان پس از ورود به سیستم می‌توانند درخواست تعمیرات ثبت کنند. این فرآیند شامل اعتبارسنجی داده‌های ورودی و آپلود تصویر الزامی است.

	فرآیند به شرح زیر است:
	
	\begin{enumerate}
		\item \textbf{پر کردن فرم}: دانشجو عنوان، توضیحات، نوع خدمات، محل مشکل و تصویر را وارد می‌کند
		
		\item \textbf{اعتبارسنجی}: سیستم داده‌ها را بررسی کرده و در صورت نامعتبر بودن، خطا نمایش می‌دهد و دانشجو فرم را اصلاح می‌کند
		
		\item \textbf{ثبت درخواست}: پس از تأیید اعتبار، سیستم درخواست را ایجاد کرده، وضعیت را به \textbf{Pending} تنظیم و تصویر را ذخیره می‌کند

	\end{enumerate}

	مدیر سیستم مسئول بررسی درخواست‌های در انتظار، تعیین اولویت و تصمیم‌گیری نهایی است.

	\begin{enumerate}

		\item \textbf{تصمیم‌گیری}: دو مسیر ممکن است:
		\begin{itemize}
			\item \textbf{تأیید}: مدیر اولویت را تعیین (\lr{High/Medium/Low}) کرده، سیستم وضعیت را به \textbf{Approved} تغییر داده و ایمیل تأیید ارسال می‌کند. درخواست برای متخصصان قابل مشاهده می‌شود

		\end{itemize}
	\end{enumerate}
	
	نکته کلیدی این فرآیند، الزام تعیین اولویت قبل از تأیید درخواست است که در پیاده‌سازی کنترل می‌شود.

	\begin{enumerate}
		\item \textbf{مشاهده و انتخاب}: متخصص درخواست‌های موجود مطابق تخصص خود را مشاهده و یکی را انتخاب می‌کند
		
		\item \textbf{دریافت درخواست}: با کلیک بر روی \textbf{Claim}، سیستم متخصص را اختصاص داده و زمان را ثبت می‌کند
		
		\item \textbf{شروع کار}: متخصص کار را آغاز کرده و یادداشت‌های فنی را وارد می‌کند. سیستم وضعیت را به \textbf{In Progress} تغییر داده و ایمیل اطلاع‌رسانی ارسال می‌کند
		
		\item \textbf{تکمیل}: متخصص یادداشت‌های نهایی و تصویر تکمیل را آپلود می‌کند. سیستم وضعیت را به \textbf{Completed} تغییر داده، تصویر را ذخیره کرده و ایمیل تکمیل به دانشجو ارسال می‌کند
		
		\item \textbf{به‌روزرسانی امتیاز}: پس از دریافت امتیاز از دانشجو، سیستم میانگین امتیاز متخصص را محاسبه و ثبت می‌کند
	\end{enumerate}
	
	این فرآیند شفافیت کامل در مراحل انجام کار و مستندسازی تصویری قبل و بعد از تعمیرات را تضمین می‌کند.

		\item \textbf{بررسی نتیجه}: دانشجو تصویر و یادداشت‌های تکمیل را بررسی کرده و تصمیم می‌گیرد
		
		\item \textbf{تصمیم‌گیری}: دو مسیر ممکن است:

	این سیستم امکان ارزیابی عینی عملکرد متخصصان و بهبود مستمر کیفیت خدمات را فراهم می‌کند. امتیازات جمع‌آوری شده در تصمیم‌گیری‌های مدیریتی و اولویت‌بندی متخصصان مورد استفاده قرار می‌گیرد.
	
	\subsection{نمودار فعالیت فرآیند بازیابی رمز عبور}

		}
		\caption{نمودار فعالیت بازیابی رمز عبور}
		\label{fig:forget_password_activity}
	\end{figure}
	
	مراحل این فرآیند عبارتند از:
	
	\begin{enumerate}
		\item \textbf{درخواست بازیابی}: کاربر بر روی \textbf{Forget Password} کلیک کرده و شناسه خود (ایمیل برای دانشجو یا کد کارمندی برای متخصص) را وارد می‌کند
		
		\item \textbf{اعتبارسنجی}: سیستم شناسه را بررسی می‌کند:
		\begin{itemize}
			\item \textbf{معتبر}: رمز تصادفی ۸ کاراکتری تولید کرده، آن را هش و در پایگاه داده ذخیره می‌کند. سپس ایمیل حاوی رمز جدید به صورت غیرهمزمان ارسال می‌شود
			
			\item \textbf{نامعتبر}: پیام خطا نمایش داده می‌شود و فرآیند متوقف می‌گردد
		\end{itemize}
		
		\item \textbf{دریافت رمز}: کاربر ایمیل را بررسی کرده، رمز جدید را دریافت و با آن وارد سیستم می‌شود
	\end{enumerate}
	
	این فرآیند از امنیت بالایی برخوردار است زیرا رمز به صورت تصادفی تولید شده، هش می‌گردد و تنها به ایمیل ثبت شده ارسال می‌شود. همچنین برای جلوگیری از شناسایی حساب‌های موجود، سیستم در صورت نامعتبر بودن شناسه نیز پیام عمومی نمایش می‌دهد.

	\begin{enumerate}
		\item \textbf{ایجاد تیکت}: کاربر عنوان و توضیحات مسئله را وارد کرده و تیکت را ارسال می‌کند. سیستم تیکت را با وضعیت \textbf{Pending} ایجاد می‌کند
		
		\item \textbf{شروع چت}: کاربر رابط چت را باز کرده و پیام می‌فرستد. سیستم از طریق \lr{WebSocket} پیام را پخش کرده، در پایگاه داده ذخیره و زمان تیکت را به‌روزرسانی می‌کند
		
		\item \textbf{پاسخ مدیر}: مدیر تیکت را مشاهده کرده و دو گزینه دارد:

			\item \textbf{بستن تیکت}: مدیر تیکت را می‌بندد و فرآیند خاتمه می‌یابد
		\end{itemize}
	\end{enumerate}
	
	این سیستم با استفاده از \lr{Django Channels} و \lr{Redis} پیاده‌سازی شده و قابلیت ارسال و دریافت پیام بدون نیاز به بارگذاری مجدد صفحه را فراهم می‌کند. تمامی پیام‌ها در پایگاه داده ذخیره شده و تاریخچه کامل مکالمات قابل دسترسی است.

		\item \textbf{محاسبه خودکار}: سیستم تعداد موارد جدید را محاسبه کرده و آخرین زمان بازدید مدیر را بررسی می‌کند

		\item \textbf{به‌روزرسانی}: سیستم زمان بازدید را ثبت کرده و تعداد نوتیفیکیشن‌ها را پاک می‌کند
	\end{enumerate}
	
	این سیستم با استفاده از \lr{AJAX} و نظرسنجی دوره‌ای پیاده‌سازی شده و نیازی به بارگذاری مجدد صفحه ندارد. موارد قابل پیگیری شامل \textbf{ثبت‌نام‌های جدید}، \textbf{درخواست‌های تعمیرات}، \textbf{تیکت‌های جدید} و \textbf{پیام‌های خوانده نشده} است.

	درخواست تعمیرات در طول چرخه عمر خود حالت‌های مختلفی را طی می‌کند که در نمودار زیر نمایش داده شده است.

		\item \textbf{Rejected}: رد درخواست توسط مدیر با ارسال دلیل

	\chapter{تکنولوژی های مورد استفاده}
	\vspace{-2.5\baselineskip}
	\noindent\rule{\textwidth}{0.4pt}
	
	پیاده‌سازی سیستم مدیریت خوابگاه مستلزم انتخاب و ترکیب مجموعه‌ای از تکنولوژی‌های مدرن و قابل اعتماد است که بتواند نیازهای عملکردی، امنیتی و قابلیت مقیاس‌پذیری این سامانه را برآورده سازد. انتخاب هر یک از این تکنولوژی‌ها بر اساس معیارهای دقیقی نظیر پایداری، امنیت، عملکرد، سهولت نگهداری و انطباق با نیازهای خاص محیط خوابگاه دانشجویی صورت گرفته است.
	
	معماری کلی سیستم بر پایه فریمورک \lr{Django} پایتون بنا شده که قابلیت‌های \lr{Real-time} را از طریق \lr{Django Channels} و \lr{WebSocket} پشتیبانی می‌کند. این انتخاب به دلیل ماهیت پیچیده تعاملات بین سه نوع کاربر مختلف (دانشجویان، متخصصان خدمات، مدیران)، نیاز به ارتباطات لحظه‌ای در قالب چت و نوتیفیکیشن‌های خودکار انجام شده است.
	
	سیستم شامل هفت لایه اصلی تکنولوژیک است: \lr{Backend} و \lr{Framework} اصلی، پایگاه داده، احراز هویت و امنیت، رابط کاربری مدیریتی، تکنولوژی‌های \lr{Frontend}، ارتباطات \lr{Real-time}، پردازش پس‌زمینه و کشینگ، سرویس‌های ارتباطی، \lr{WebServer} و \lr{Deployment} و کنترل نسخه. هر یک از این لایه‌ها با ابزارها و تکنولوژی‌های تخصصی پیاده‌سازی شده‌اند که به صورت یکپارچه و هماهنگ با یکدیگر عمل کرده و تجربه کاربری روان و کارآمدی را فراهم می‌سازند.
	
	در ادامه این فصل، هر یک از تکنولوژی‌های مورد استفاده به تفصیل معرفی شده و دلایل انتخاب، مزایا، چالش‌های پیاده‌سازی و نحوه ادغام آن‌ها در معماری کلی سیستم تحلیل می‌شود.
	
	\section{چارچوب اصلی و بک‌اند}
	\subsection{\lr{Django}}
	\lr{Django} یک فریم‌ورک وب پایتونی با معماری \lr{Model-View-Template (MVT)} است که بر اصل \lr{Don't Repeat Yourself (DRY)} بنا شده است. این فریم‌ورک شامل مجموعه‌ای از ابزارها و کتابخانه‌های از پیش ساخته شده است که فرآیند توسعه وب را تسریع می‌بخشد.
	
	\textbf{ویژگی‌های کلیدی \lr{Django} در پروژه:}
	
	\textbf{\lr{Object-Relational Mapping (ORM)}:} سیستم \lr{ORM} جنگو امکان تعریف مدل‌های داده به صورت کلاس‌های پایتونی و تعامل با پایگاه داده بدون نوشتن کوئری‌های \lr{SQL} خام را فراهم می‌کند. در این پروژه مدل‌هایی نظیر \lr{Student}، \lr{ServiceExpert}، \lr{MaintenanceRequest} و \lr{Ticket} با روابط پیچیده بین جداول تعریف شده‌اند.
	
	\textbf{\lr{Admin Interface}:} پنل مدیریتی خودکار جنگو برای مدیریت داده‌ها استفاده شده که با \lr{Jazzmin} شخصی‌سازی گردیده است. این پنل امکانات \lr{CRUD} کامل، فیلترینگ پیشرفته و اقدامات گروهی را برای مدیریت درخواست‌ها، کاربران و متخصصان فراهم می‌کند.
	
	\textbf{\lr{Authentication System}:} سیستم احراز هویت داخلی جنگو برای مدیریت کاربران به کار رفته که با سیستم‌های احراز هویت سفارشی برای دانشجویان (با شماره دانشجویی) و متخصصان (با کد کارمندی) توسعه یافته است.
	
	\textbf{\lr{Forms Framework}:} سیستم فرم‌های جنگو برای ایجاد، اعتبارسنجی و پردازش فرم‌های مختلف نظیر ثبت‌نام دانشجویان، ثبت درخواست تعمیرات و امتیازدهی استفاده شده است.
	
	\subsection{\lr{Django Signals}}
	سیگنال‌های جنگو مکانیزم اصلی برای پیاده‌سازی \lr{Event-Driven Programming} در این پروژه هستند. این سیستم امکان اجرای خودکار عملیات مرتبط پس از وقوع رویدادهای خاص را فراهم می‌کند.
	
	\textbf{کاربردهای \lr{Signals} در پروژه:}
	
	\textbf{\lr{Post-save Signals}:} برای ارسال ایمیل‌های خودکار پس از تغییر وضعیت درخواست‌ها استفاده می‌شود. زمانی که وضعیت \lr{MaintenanceRequest} از \lr{pending} به \lr{approved} تغییر می‌کند، سیگنال مربوطه ایمیل اطلاع‌رسانی را به دانشجو ارسال می‌کند.
	
	\textbf{\lr{Pre-save Signals}:} برای اعتبارسنجی داده‌ها قبل از ذخیره استفاده شده است. برای مثال، بررسی تنظیم اولویت درخواست قبل از تأیید نهایی.
	
	\textbf{\lr{Post-init Signals}:} برای ضبط وضعیت اولیه مدل‌ها جهت مقایسه با تغییرات بعدی به کار می‌رود.
	
	\textbf{مزایای استفاده از \lr{Signals}:} جداسازی منطق کسب‌وکار، امکان توسعه آسان ویژگی‌های جدید بدون تغییر کد اصلی و اطمینان از اجرای عملیات مرتبط به صورت خودکار.
	
	\subsection{\lr{ASGI (Asynchronous Server Gateway Interface)}}
	\lr{ASGI} پروتکل جانشین \lr{WSGI} برای پشتیبانی از عملیات ناهمزمان در اپلیکیشن‌های پایتونی است. این پروتکل امکان مدیریت همزمان انواع مختلف ارتباطات شامل \lr{HTTP}، \lr{WebSocket} و سایر پروتکل‌های ناهمزمان را فراهم می‌کند.
	
	\textbf{پیاده‌سازی \lr{ASGI} در پروژه:}
	
	\textbf{پشتیبانی \lr{WebSocket}:} برای پیاده‌سازی چت \lr{Real-time} بین کاربران و مدیریت استفاده شده است. کاربران می‌توانند بدون بارگذاری مجدد صفحه پیام‌ها را ارسال و دریافت کنند.
	
	\textbf{پردازش ناهمزمان:} پردازش درخواست‌های \lr{HTTP} معمولی در کنار مدیریت اتصالات \lr{WebSocket} به صورت همزمان انجام می‌شود.
	
	\textbf{لایه‌های کانال:} با استفاده از \lr{Redis} به عنوان \lr{MessageBroker}، پیام‌ها بین اتصالات مختلف \lr{WebSocket} و \lr{instance} های مختلف اپلیکیشن منتقل می‌شوند.
	
	\textbf{\lr{ASGI Benefit}:} قابلیت پردازش همزمان درخواست‌های متعدد، پشتیبانی از ارتباطات \lr{Real-time}، مقیاس‌پذیری بهتر در مقایسه با \lr{WSGI} و امکان تعامل دوطرفه بین سرور و کلاینت.
	
	\textbf{تنظیمات \lr{ASGI}:} فایل \lr{asgi.py} پروژه برای مسیریابی ترافیک \lr{HTTP} و \lr{WebSocket} تنظیم شده و با \lr{Django Channels} یکپارچه سازی گردیده است.
	
	
	\section{\lr{PostgreSQL}}
	\lr{PostgreSQL} یک سیستم مدیریت پایگاه داده رابطه‌ای (\lr{RDBMS}) پیشرفته و متن باز است که برای پروژه‌های مقیاس بزرگ و حرفه‌ای طراحی شده است. انتخاب این پایگاه داده برای سیستم مدیریت خوابگاه بر اساس نیازهای خاص پروژه و ویژگی‌های منحصر به فرد آن صورت گرفته است.
	
	\textbf{ویژگی‌های کلیدی \lr{PostgreSQL}:}
	
	\textbf{پشتیبانی از انواع داده پیچیده:} \lr{PostgreSQL} قابلیت ذخیره و پردازش انواع مختلف داده از جمله \lr{JSON}، \lr{JSONB}، \lr{Arrays} و \lr{Custom Types} را داراست. در این پروژه از فیلدهای \lr{JSON} برای ذخیره اطلاعات پیچیده نظیر تنظیمات کاربری و متادیتای تصاویر استفاده شده است.
	
	\textbf{رعایت کامل اصول \lr{ACID}:} تمامی تراکنش‌ها در \lr{PostgreSQL} دارای ویژگی‌های \lr{Atomicity}، \lr{Consistency}، \lr{Isolation} و \lr{Durability} هستند که برای سیستمی با تعاملات پیچیده بین دانشجویان، متخصصان و مدیران ضروری است.
	
	\textbf{قابلیت مقیاس‌پذیری بالا:} \lr{PostgreSQL} توانایی مدیریت حجم بالای داده و کاربران همزمان را داراست که برای سیستم خوابگاه با صدها دانشجو و درخواست‌های متعدد روزانه مناسب است.
	
	\textbf{عملکرد بهینه:} موتور کوئری پیشرفته \lr{PostgreSQL} با قابلیت‌های بهینه‌سازی خودکار کوئری‌ها و \lr{indexing} هوشمند سرعت دسترسی به داده‌ها را افزایش می‌دهد.
	
	\section{احراز هویت و امنیت}
	امنیت و مدیریت دسترسی‌ها از مهم‌ترین بخش‌های هر سامانه نرم‌افزاری محسوب می‌شود. در سیستم مدیریت خوابگاه، ترکیبی از قابلیت‌های داخلی \lr{Django} و پروتکل‌های استاندارد امنیتی وب برای تضمین امنیت داده‌ها و کاربران به کار گرفته شده است. این بخش شامل دو بخش اصلی است: \textbf{\lr{Django Auth}} برای مدیریت کاربران و کنترل دقیق سطح دسترسی‌ها و \textbf{\lr{SSL/TLS}} برای رمزنگاری ارتباطات و محافظت از داده‌ها در بستر اینترنت. در ادامه به معرفی و نقش هر کدام پرداخته می‌شود.
	
	\subsection{\lr{Django Auth}}
	سیستم احراز هویت \lr{Django} یکی از قدرتمندترین و امن‌ترین راهکارهای داخلی برای مدیریت کاربران در فریمورک جنگو است. این سیستم به عنوان هسته مرکزی امنیت سامانه مدیریت خوابگاه عمل می‌کند و امکان کنترل دقیق دسترسی‌ها را فراهم می‌آورد.
	
	\textbf{ویژگی‌های اصلی سیستم احراز هویت:}
	
	\textbf{مدیریت کاربران:} \lr{Django Auth} سیستم جامعی برای ایجاد و مدیریت کاربران ارائه می‌دهد که شامل فیلدهای اساسی نظیر نام کاربری، رمز عبور، ایمیل و وضعیت فعالیت است. در این پروژه مدل \lr{User} سفارشی‌سازی شده تا علاوه بر فیلدهای پیش‌فرض، شامل نوع کاربر (دانشجو یا متخصص) و شماره تماس نیز باشد.
	
	\textbf{هش کردن رمزهای عبور:} تمامی رمزهای عبور با استفاده از الگوریتم \lr{PBKDF2} همراه با \lr{SHA256} به صورت امن هش می‌شوند و هرگز به صورت متن خام در پایگاه داده ذخیره نمی‌گردند. این روش از حملات \lr{Rainbow Table} و \lr{Brute Force} محافظت می‌کند.
	
	\textbf{سیستم مجوزها و گروه‌ها:} \lr{Django} امکان تعریف مجوزهای سفارشی و گروه‌بندی کاربران را فراهم می‌کند. در پروژه سه گروه اصلی (\lr{Students}، \lr{ServiceExperts}، \lr{Administrators}) با مجوزهای متفاوت تعریف شده‌اند.
	
	\textbf{بکند احراز هویت سفارشی:} برای پاسخ به نیازهای خاص پروژه دو بکند سفارشی توسعه یافته است. \lr{StudentNumberBackend} که امکان ورود دانشجویان با شماره دانشجویی ۹ رقمی را فراهم می‌کند و \lr{ServiceExpertEmployeeIdBackend} برای ورود متخصصان با کد کارمندی.
	
	\textbf{مدیریت نشست‌ها:} سیستم مدیریت نشست‌ها شامل قابلیت‌های پیشرفته‌ای نظیر تنظیم زمان انقضای نشست، \lr{Remember Me} برای نگهداری نشست‌های طولانی‌مدت و امنیت کوکی‌ها با تنظیمات \lr{HttpOnly} و \lr{Secure} است.
	
	\textbf{پیاده‌سازی سیستم‌های احراز هویت چندگانه:}
	
	\textbf{احراز هویت دانشجویان:} دانشجویان می‌توانند با شماره دانشجویی ۹ رقمی و رمز عبور وارد سیستم شوند. فرآیند شامل اعتبارسنجی فرمت شماره، تطبیق با رکوردهای موجود، بررسی وضعیت تأیید و فعالیت حساب است.
	
	\textbf{احراز هویت متخصصان:} متخصصان خدمات با استفاده از کد کارمندی منحصربه‌فرد و رمز عبور دسترسی پیدا می‌کنند. سیستم وضعیت فعالیت متخصص و انطباق تخصص با درخواست‌ها را نیز کنترل می‌کند.
	
	\textbf{احراز هویت مدیران:} مدیران سیستم با ایمیل و رمز عبور وارد پنل مدیریتی می‌شوند و دسترسی کامل به تمامی بخش‌های سیستم دارند.
	
	\subsection{\lr{HTTPS} برای \lr{SSL}}
	پروتکل \lr{SSL/TLS} لایه امنیتی اصلی برای محافظت از ارتباطات بین کلاینت و سرور است که تمامی داده‌های منتقل شده را رمزگذاری می‌کند.
	
	\textbf{مزایای پیاده‌سازی \lr{SSL}:}
	
	\textbf{رمزگذاری داده‌ها:} تمامی اطلاعات منتقل شده بین مرورگر کاربران و سرور شامل رمزهای عبور، اطلاعات شخصی، تصاویر درخواست‌ها و پیام‌های چت به صورت کامل رمزگذاری می‌شوند.
	
	\textbf{احراز هویت سرور:} گواهی \lr{SSL} هویت سرور را تأیید می‌کند و از حملات \lr{Man-in-the-Middle} جلوگیری می‌کند. کاربران با مشاهده قفل سبز در مرورگر از امنیت اتصال مطمئن می‌شوند.
	
	\textbf{یکپارچگی داده‌ها:} \lr{SSL} اطمینان حاصل می‌کند که داده‌ها در طول انتقال تغییر نکرده‌اند و از حملات \lr{Data Tampering} محافظت می‌کند.
	
	\textbf{پیکربندی امنیتی پیشرفته:}
	
	\textbf{کوکی‌های امن:} تمامی کوکی‌های حساس با فلگ \lr{Secure} و \lr{HttpOnly} تنظیم شده‌اند تا از \lr{XSS} و \lr{Session Hijacking} محافظت کنند.
	
	\section{رابط مدیریت}
	
	رابط مدیریت بخش مهمی از هر سامانه نرم‌افزاری است که امکان کنترل و نظارت کامل بر داده‌ها و فرآیندهای سیستم را برای مدیران فراهم می‌کند. در این پروژه از \lr{Django Admin} به عنوان هسته اصلی مدیریت داده‌ها و از \lr{Jazzmin} برای بهبود ظاهر و تجربه کاربری این پنل استفاده شده است. این ترکیب باعث شده است علاوه بر داشتن امکانات قدرتمند مدیریتی، رابطی زیبا، کاربرپسند و قابل شخصی‌سازی نیز در اختیار مدیران قرار گیرد. در ادامه به ویژگی‌ها و نقش هر کدام پرداخته می‌شود.
	
	\subsection{\lr{Django Admin}}
	
	پنل مدیریت \lr{Django} یکی از قدرتمندترین و کاربردی‌ترین ویژگی‌های این فریمورک است که امکان مدیریت کامل داده‌ها و عملیات سیستم را بدون نیاز به توسعه رابط کاربری جداگانه فراهم می‌کند. این پنل به عنوان مرکز کنترل اصلی سیستم مدیریت خوابگاه عمل می‌کند.
	
	\textbf{ویژگی‌های اصلی \lr{Django Admin} در پروژه:}
	
	\textbf{مدیریت خودکار مدل‌ها:} پنل ادمین به صورت خودکار از مدل‌های تعریف شده در پروژه نظیر \lr{Student}، \lr{ServiceExpert}، \lr{MaintenanceRequest} و \lr{Ticket} رابط‌های \lr{CRUD} کامل ایجاد می‌کند. هر مدل با تنظیمات سفارشی در کلاس‌های \lr{Admin} مربوطه پیکربندی شده است.
	
	\textbf{سیستم فیلترینگ و جستجو:} برای هر مدل فیلترهای مخصوص طراحی شده که امکان جستجو بر اساس وضعیت، تاریخ، نوع کاربر و سایر معیارها را فراهم می‌کند. برای مثال درخواست‌های تعمیرات بر اساس وضعیت، اولویت، نوع خدمات و ساختمان قابل فیلتر هستند.
	
	\textbf{اقدامات گروهی:} قابلیت‌های پیشرفته‌ای برای انجام عملیات روی چندین رکورد همزمان تعریف شده است. مدیران می‌توانند درخواست‌ها را به صورت دسته‌ای تأیید یا رد کنند، اولویت‌ها را تغییر دهند و متخصصان را خودکار اختصاص دهند.
	
	\textbf{نمایش‌های سفارشی:} لیست‌های نمایش هر مدل با ستون‌های اختصاصی طراحی شده‌اند. برای مثال لیست درخواست‌ها شامل عنوان مختصر، اطلاعات دانشجو، نوع خدمات، اولویت، وضعیت، متخصص مسئول، امتیاز دانشجو و روزهای گذشته از ثبت است.
	
	\textbf{ویرایش درون‌خطی:} امکان ویرایش مدل‌های مرتبط در همان صفحه فراهم شده است. برای مثال هنگام ویرایش درخواست تعمیرات می‌توان تصاویر مربوطه را نیز در همان صفحه مدیریت کرد.
	
	\textbf{پیکربندی‌های امنیتی:}
	
	\textbf{کنترل دسترسی:} سیستم مجوزهای دقیق برای کنترل دسترسی مدیران مختلف به بخش‌های مختلف تعریف شده است. برخی مدیران تنها به مدیریت درخواست‌ها و برخی به مدیریت کاربران دسترسی دارند.
	
	\textbf{تاریخچه تغییرات:} تمامی تغییرات انجام شده روی رکوردها شامل زمان تغییر، کاربر انجام‌دهنده و جزئیات تغییرات ثبت و قابل پیگیری است.
	
	\textbf{اعتبارسنجی:} اعتبارسنجی‌های سفارشی برای جلوگیری از ورود داده‌های نامعتبر تعریف شده است. برای مثال عدم امکان تأیید درخواست‌هایی که اولویت آنها تعیین نشده است.
	
	\subsection{\lr{Jazzmin}}
	
	\lr{Jazzmin} یک پکیج تم‌سازی پیشرفته برای \lr{Django Admin} است که ظاهر و تجربه کاربری پنل مدیریت را به طور کامل متحول می‌کند. این افزونه بدون تغییر در عملکرد اصلی \lr{Admin} امکانات بصری و کاربری پیشرفته‌ای ارائه می‌دهد.
	
	\textbf{بهبودهای بصری \lr{Jazzmin}:}
	
	\textbf{طراحی مدرن و ریسپانسیو:} رابط کاربری به سبک مدرن با پشتیبانی کامل از دستگاه‌های موبایل و تبلت طراحی شده است. منوهای کشویی، نوارهای پیمایش و کارت‌های اطلاعاتی بهینه‌سازی شده‌اند.
	
	\textbf{شخصی‌سازی رنگ‌ها:} پالت رنگی سامانه مطابق با هویت بصری دانشگاه و خوابگاه تنظیم شده است. رنگ‌های اصلی، فرعی و حالت‌های تعاملی قابل تنظیم هستند.
	
	\textbf{آیکون‌گذاری هوشمند:} هر بخش از سیستم با آیکون‌های \lr{FontAwesome} مناسب نمایش داده می‌شود. درخواست‌ها با آیکون ابزار، دانشجویان با آیکون کاربر و تیکت‌ها با آیکون پیام مشخص می‌شوند.
	
	\textbf{منوهای پویا:} سیستم منوی چندسطحی با قابلیت باز و بسته شدن برای دسترسی آسان‌تر به بخش‌های مختلف طراحی شده است.
	
	\textbf{ویژگی‌های عملکردی پیشرفته:}
	
	\textbf{داشبورد سفارشی:} صفحه اصلی پنل ادمین با کارت‌های آماری برای نمایش وضعیت کلی سیستم طراحی شده است. آمار درخواست‌های جدید، دانشجویان ثبت‌نام شده و تیکت‌های باز در کارت‌های جداگانه نمایش داده می‌شود.
	
	\textbf{نوار اطلاع‌رسانی:} سیستم نوتیفیکیشن پیشرفته در بالای صفحه که هر ۱۰ ثانیه وضعیت موارد جدید را بررسی می‌کند. درخواست‌های جدید، پیام‌های خوانده نشده و ثبت‌نام‌های در انتظار تأیید نمایش داده می‌شوند.
	
	\textbf{فرم‌های بهبود یافته:} فرم‌های ویرایش و ایجاد رکورد با المان‌های تعاملی بهتر نظیر \lr{Date Picker}، \lr{Select2} و \lr{File Upload} با پیش‌نمایش تجهیز شده‌اند.
	
	\textbf{سرعت و عملکرد:} بهینه‌سازی‌هایی برای کاهش زمان بارگذاری صفحات و بهبود تجربه کاربری در سیستم‌های با حجم داده بالا انجام شده است.
	
	\textbf{تنظیمات اختصاصی پروژه:}
	
	\textbf{سفارشی‌سازی برند:} لوگو و نام سیستم مدیریت خوابگاه در تمام صفحات نمایش داده می‌شود.
	
	\textbf{پیکربندی چیدمان:} چیدمان صفحات بر اساس نیازهای مدیران خوابگاه تنظیم شده و امکان تغییر آسان در آینده وجود دارد.
	
	\textbf{پشتیبانی چندزبانه:} پشتیبانی از زبان فارسی با چیدمان \lr{RTL} برای سازگاری کامل با محتوای فارسی.
	
	\section{تکنولوژی‌های طرف کلاینت}
	بخش فرانت‌اند مسئول نمایش و تعامل مستقیم با کاربر است. تمام آنچه کاربر در صفحه وب می‌بیند و با آن کار می‌کند، توسط تکنولوژی‌های فرانت‌اند ساخته می‌شود. این تکنولوژی‌ها شامل ابزارها و زبان‌هایی هستند که برای ایجاد ساختار، طراحی و تعامل در صفحات وب به کار می‌روند. در این پروژه برای پیاده‌سازی رابط کاربری سیستم از سه تکنولوژی اصلی یعنی \textbf{\lr{HTML}}، \textbf{\lr{CSS}} و \textbf{\lr{JavaScript}} استفاده شده است که در ادامه توضیح داده می‌شوند.
	
	\textbf{\lr{HTML (HyperText Markup Language)}:} زبان اصلی برای ساخت صفحات وب است. مثل یک چارچوب یا اسکلت صفحه عمل می‌کند که محتوا را سازماندهی می‌کند. در این پروژه از \lr{HTML} برای ساخت فرم‌های ثبت‌نام و ورود دانشجویان، نمایش جداول درخواست‌ها و اطلاعات متخصصان، ایجاد صفحات مختلف سیستم مثل داشبورد و گزارش‌ها و آپلود تصاویر و فایل‌ها استفاده شده است.
	
	\textbf{\lr{CSS (Cascading Style Sheets)}:} برای زیبا کردن صفحات وب استفاده می‌شود. تمام رنگ‌ها، فونت‌ها، چیدمان و ظاهر سیستم با \lr{CSS} طراحی شده است. این شامل طراحی ظاهر کلی سیستم با رنگ‌های مناسب، ایجاد دکمه‌های زیبا و منوهای کاربردی، تنظیم نمایش در موبایل، تبلت و کامپیوتر و افکت‌های ساده مثل تغییر رنگ دکمه‌ها هنگام کلیک می‌باشد.
	
	\textbf{\lr{JavaScript}:} برای تعامل و پویایی صفحات استفاده می‌شود. بدون آن صفحات ثابت و غیرقابل تعامل می‌مانند. در پروژه از \lr{JavaScript} برای بررسی صحت اطلاعات قبل از ارسال فرم‌ها، نمایش پیام‌های موفقیت و خطا، تغییر محتوای صفحه بدون بارگذاری مجدد، کنترل منوها و بخش‌های مختلف صفحه و مدیریت تایمر ۱۵ دقیقه‌ای برای تأیید ایمیل استفاده شده است.
	
	\section{ارتباطات لحظه‌ای}
	
	ارتباطات لحظه‌ای یکی از پیچیده‌ترین و حیاتی‌ترین بخش‌های سیستم مدیریت خوابگاه محسوب می‌شود که امکان تعامل فوری و دوطرفه بین کاربران و مدیریت را فراهم می‌آورد. این بخش مسئول پیاده‌سازی سیستم چت آنلاین، نوتیفیکیشن‌های خودکار و به‌روزرسانی لحظه‌ای وضعیت درخواست‌ها است. معماری این بخش بر اساس تکنولوژی‌های مدرن و استاندارد وب طراحی شده تا بتواند تعداد زیادی اتصال همزمان را مدیریت کرده و تجربه کاربری روان و پایداری ارائه دهد.
	
	\subsection{\lr{Django Channels}}
	
	\lr{Django Channels} یک افزودنی برای جنگو است که امکان پردازش درخواست‌های همزمان (\lr{asynchronous}) را فراهم می‌کند. با استفاده از \lr{Channels} می‌توان علاوه بر \lr{HTTP}، وب سوکت‌ها، \lr{long-polling} و سایر پروتکل‌های بلادرنگ را مدیریت کرد. این ابزار برای ایجاد چت آنلاین، نوتیفیکیشن لحظه‌ای و بازی‌های تحت وب بسیار کاربردی است.
	
	\textbf{ویژگی‌های پیشرفته \lr{Django Channels} در پروژه:}
	
	\textbf{معماری رویداد محور:} بر اساس مفهوم رویداد محور عمل می‌کند که هر عملیات کاربر نظیر ارسال پیام، تغییر وضعیت درخواست یا ورود به سیستم رویدادی تولید می‌کند که سایر بخش‌های سیستم می‌توانند به آن واکنش نشان دهند. این معماری امکان جداسازی منطق کسب‌وکار و پردازش رویدادها را فراهم می‌آورد.
	
	\textbf{پشتیبانی از چندین کارگر:} سیستم قابلیت اجرای چندین کارگر همزمان را داراست که هر کارگر می‌تواند مجموعه‌ای از اتصالات \lr{WebSocket} را مدیریت کند. این قابلیت امکان مقیاس‌پذیری افقی و تحمل‌پذیری در برابر خرابی را فراهم می‌کند.
	
	\textbf{مدیریت حافظه بهینه:} \lr{Channels} سیستم مدیریت حافظه هوشمندی دارد که اتصالات غیرفعال را به صورت خودکار پاک کرده و از انباشت حافظه جلوگیری می‌کند. همچنین قابلیت تنظیم محدودیت تعداد اتصالات همزمان برای هر کاربر نیز وجود دارد.
	
	\textbf{سیستم احراز هویت یکپارچه:} \lr{Channels} به صورت یکپارچه با سیستم احراز هویت \lr{Django} کار می‌کند و امکان تشخیص کاربر در اتصالات \lr{WebSocket}، کنترل دسترسی بر اساس نقش کاربر و مدیریت نشست‌ها در ارتباطات \lr{real-time} را فراهم می‌آورد.
	
	\subsection{\lr{WebSocket}}
	
	\lr{WebSocket} یک پروتکل ارتباطی دوطرفه و دائم بین مرورگر و سرور است. برخلاف \lr{HTTP} که هر بار یک درخواست و پاسخ دارد، \lr{WebSocket} پس از اتصال اولیه، امکان ارسال و دریافت داده را در لحظه و بدون نیاز به بارگذاری مجدد صفحه فراهم می‌کند. این پروتکل برای چت، نوتیفیکیشن، بازی آنلاین و سایر کاربردهای \lr{real-time} ایده‌آل است.
	
	\textbf{پیاده‌سازی \lr{WebSocket} در سیستم:}
	
	\textbf{مدیریت چرخه حیات اتصال:} سیستم \lr{WebSocket} شامل سه مرحله اصلی است که مرحله دست‌دادن برای برقراری اتصال اولیه و احراز هویت کاربر، مرحله ارتباط برای تبادل پیام‌های دوطرفه و پردازش درخواست‌های مختلف و مرحله خاتمه برای قطع اتصال کنترل شده و پاک‌سازی منابع می‌باشد.
	
	\textbf{پروتکل پیام‌رسانی ساختاریافته:} تمامی پیام‌های ارسالی در قالب \lr{JSON} با فیلدهای استاندارد شامل \lr{type} برای نوع پیام، \lr{message} برای محتوای اصلی، \lr{timestamp} برای زمان ارسال، \lr{sender} برای شناسایی فرستنده و \lr{metadata} برای اطلاعات تکمیلی تنظیم شده‌اند.
	
	\textbf{مکانیزم ضربان قلب:} برای اطمینان از سلامت اتصال، سیستم ضربان قلب پیاده‌سازی شده که هر ۳۰ ثانیه پیام \lr{ping/pong} بین کلاینت و سرور رد و بدل می‌شود. در صورت عدم دریافت پاسخ در زمان مشخص، اتصال به صورت خودکار قطع و مجدداً برقرار می‌گردد.
	
	\textbf{کنترل کیفیت پیام‌ها:} سیستم قابلیت کنترل طول پیام‌ها، فیلتر کردن محتوای نامناسب، محدود کردن سرعت ارسال پیام (\lr{rate limiting}) و تشخیص پیام‌های تکراری را داراست تا از سوءاستفاده و \lr{spam} جلوگیری کند.
	
	\subsection{نظرسنجی و تازه‌سازی خودکار}
	
	نظرسنجی یا تازه‌سازی خودکار روشی برای به‌روزرسانی داده‌ها در صفحات وب است که مرورگر به صورت دوره‌ای از سرور درخواست جدید می‌فرستد. این روش ساده است اما مصرف منابع بالاتری دارد و تأخیر بین بروزرسانی‌ها بیشتر از \lr{WebSocket} است. معمولاً برای داده‌های کمتر حساس به تأخیر یا به عنوان جایگزین ساده وب سوکت استفاده می‌شود.
	
	\textbf{استراتژی‌های نظرسنجی در پروژه:}
	
	\textbf{نظرسنجی کوتاه برای نوتیفیکیشن‌ها:} سیستم نوار اطلاع‌رسانی مدیریت هر ۱۰ ثانیه وضعیت جدیدترین موارد را از سرور درخواست می‌کند. این بازه زمانی بین سرعت بروزرسانی و مصرف منابع سرور تعادل ایجاد کرده و برای کاربرد مدیریتی مناسب است.
	
	\textbf{نظرسنجی انطباقی:} سیستم قابلیت تنظیم خودکار فاصله زمانی بین درخواست‌ها بر اساس میزان فعالیت کاربر داراست. در صورت فعالیت بالا، فاصله کاهش یافته و در مواقع بی‌فعالیت، فاصله افزایش می‌یابد تا مصرف منابع بهینه شود.
	
	\textbf{مکانیزم پشتیبان:} در صورت عدم پشتیبانی مرورگر از \lr{WebSocket} یا مشکلات شبکه، سیستم به صورت خودکار به نظرسنجی تغییر حالت می‌دهد تا خدمات همچنان قابل استفاده باقی بمانند.
	
	\textbf{مدیریت کش:} برای کاهش بار سرور، نتایج درخواست‌های نظرسنجی در کش محلی ذخیره شده و تنها در صورت تغییر داده‌ها، رابط کاربری بروزرسانی می‌شود.
	\section{پردازش پس‌زمینه و کشینگ}
	بخش پردازش پس‌زمینه و کشینگ نقش بحرانی در عملکرد و قابلیت اطمینان سیستم مدیریت خوابگاه ایفا می‌کند. این بخش مسئول مدیریت وظایف زمان‌بر، بهینه‌سازی سرعت پاسخ‌دهی سیستم و کاهش بار پردازشی سرور اصلی است. دو تکنولوژی اصلی این بخش شامل \lr{Celery} برای پردازش وظایف پس‌زمینه و \lr{Redis} برای کشینگ و پشتیبانی از \lr{Celery} می‌باشند.
	
	\subsection{\lr{Celery}}
	\lr{Celery} یک سیستم صف‌بندی و پردازش وظایف پس‌زمینه (\lr{background tasks}) برای پایتون است. با \lr{Celery} می‌توان کارهای زمان‌بر مثل ارسال ایمیل، پردازش فایل یا محاسبات سنگین را به صورت غیرهمزمان اجرا کرد تا درخواست‌های وب سریع‌تر پاسخ داده شوند. اغلب با \lr{Redis} یا \lr{RabbitMQ} به عنوان کارگزار برای مدیریت صف‌ها استفاده می‌شود.
	
	\textbf{معماری پردازش وظایف:}
	
	\textbf{انواع وظایف در سیستم:} وظایف پس‌زمینه سیستم در چهار دسته اصلی تقسیم شده‌اند که شامل وظایف ایمیل برای ارسال ایمیل‌های اطلاع‌رسانی، وظایف پردازش تصویر برای پردازش تصاویر آپلود شده، وظایف تجزیه و تحلیل داده‌ها برای محاسبه آمارها و تولید گزارش‌ها و وظایف نگهداری برای پاک‌سازی داده‌های موقت و عملیات نگهداری دوره‌ای می‌باشد.
	
	\textbf{ادغام سرویس ایمیل:} سه کلاس اصلی برای مدیریت ایمیل‌ها توسعه یافته که \lr{StudentEmailService} برای ایمیل‌های دانشجویان، \lr{ServiceEmailService} برای متخصصان، \lr{MaintenanceEmailService} برای اطلاع‌رسانی مراحل درخواست تعمیرات و \lr{TicketEmailService} برای اطلاع‌رسانی پیام‌های تیکت می‌باشند.
	
	\textbf{مدیریت کارگران و صف‌های اولویت:} تعداد کارگران \lr{Celery} بر اساس نوع وظیفه و بار سیستم تنظیم می‌شود. سیستم صف‌بندی در سه سطح اولویت عمل می‌کند: اولویت بالا برای وظایف حیاتی، متوسط برای وظایف معمول و پایین برای وظایف نگهداری که در زمان کم‌ترافیک اجرا می‌شوند.
	
	\textbf{مکانیزم تلاش مجدد و مدیریت خطا:} سیستم تلاش مجدد هوشمندی با عقب‌نشینی نمایی، حداکثر تلاش مجدد و تشخیص نوع خطا طراحی شده است. همچنین سیستم مانیتورینگ و هشدار برای نظارت بر وضعیت وظایف و هشدار در مواقع بروز مشکل فعال است.
	
	
	\subsection{\lr{Redis}}
	\lr{Redis} یک پایگاه داده در حافظه (\lr{in-memory}) و متن باز است که برای ذخیره‌سازی سریع داده‌ها و کشینگ استفاده می‌شود. در این پروژه \lr{Redis} نقش دوگانه دارد: به عنوان کارگزار پیام برای \lr{Celery} و به عنوان لایه کش برای بهینه‌سازی عملکرد سیستم. این ابزار سرعت بسیار بالایی نسبت به پایگاه داده‌های معمولی دارد.
	
	\textbf{نقش \lr{Redis} به عنوان کارگزار \lr{Celery}:} به عنوان واسط انتقال پیام بین \lr{Django} و کارگران \lr{Celery} عمل می‌کند. تمامی وظایف ارسالی از \lr{Django} در صف‌های \lr{Redis} ذخیره شده و کارگران آنها را به ترتیب اولویت پردازش می‌کنند. همچنین نتایج وظایف در \lr{Redis} کش شده تا دسترسی سریع فراهم شود.
	
	\textbf{استراتژی‌های کشینگ:} سیستم کشینگ چهار سطح دارد که شامل کش سطح اول برای داده‌های پرتکرار، کش سطح دوم برای نتایج کوئری‌های پیچیده، کش سطح سوم برای داده‌های استاتیک و کش موقت برای نشست‌ها می‌باشد. هر سطح دارای \lr{TTL} مخصوص خود است.
	
	
	\section{خدمات ارتباطی}
	\lr{SMTP} یک پروتکل استاندارد برای ارسال ایمیل است که به برنامه‌ها و وب‌سایت‌ها امکان ارسال ایمیل‌های خودکار مانند تأیید ثبت نام، بازیابی رمز عبور یا نوتیفیکیشن‌ها را به کاربران می‌دهد. در این پروژه سیستم از سرور \lr{SMTP} جیمیل با پورت ۵۸۷ و رمزنگاری \lr{TLS} استفاده می‌کند که امنیت بالا و قابلیت اطمینان مناسبی ارائه می‌دهد. تنظیمات شامل \lr{EMAIL\_HOST='smtp.gmail.com'}، \lr{EMAIL\_PORT=587}، \lr{EMAIL\_USE\_TLS=True} و احراز هویت با نام کاربری و رمز عبور اختصاصی است.
	
	تمامی ایمیل‌های سیستم بر اساس قالب‌های از پیش طراحی شده ارسال می‌شوند که شامل قالب‌ها برای تأیید ایمیل دانشجویان، اطلاع‌رسانی تأیید یا رد درخواست‌های ثبت‌نام، اطلاع‌رسانی تغییر وضعیت درخواست‌های تعمیرات، بازنشانی رمز عبور و پاسخ‌های مدیریت به تیکت‌ها می‌باشد.
	
	پیاده‌سازی شامل چهار کلاس سرویس اختصاصی است که \lr{StudentEmailService} برای مدیریت ایمیل‌های دانشجویان، \lr{ServiceEmailService} برای ایمیل‌های متخصصان، \lr{MaintenanceEmailService} برای اطلاع‌رسانی مراحل تعمیرات و \lr{TicketEmailService} برای اطلاع‌رسانی پیام‌های تیکت را پوشش می‌دهد.
	
	تمامی عملیات ارسال ایمیل از طریق وظایف \lr{Celery} به صورت غیرهمزمان انجام می‌شود تا سرعت پاسخ‌دهی سیستم حفظ شود و هر وظیفه دارای مکانیزم تلاش مجدد با حداکثر ۵ تلاش مجدد و فاصله زمانی عقب‌نشینی نمایی است که در صورت بروز خطا در ارسال، سیستم قادر به تلاش مجدد بوده و در نهایت در صورت ناکامی کامل، خطا در لاگ ثبت می‌شود.
	
	\section{وب سرور و استقرار}
	بخش وب سرور و استقرار مسئول آماده‌سازی، مستقرسازی و اجرای سیستم مدیریت خوابگاه در محیط تولید (\lr{production}) است. این بخش شامل مجموعه‌ای از تکنولوژی‌ها و ابزارهایی است که سیستم را قادر می‌سازد تا بتواند ترافیک بالا را مدیریت کند، امنیت مناسب داشته باشد و در دسترس بودن (\lr{availability}) بالایی را تضمین نماید. سه تکنولوژی اصلی این بخش شامل \lr{Nginx} به عنوان وب سرور اصلی و پراکسی معکوس، \lr{Gunicorn} به عنوان سرور \lr{WSGI} برای اجرای اپلیکیشن \lr{Django} و \lr{Docker} برای کانتینرسازی و مدیریت آسان محیط‌های مختلف می‌باشند.
	
	\subsection{\lr{Nginx}}
	یک وب سرور و پراکسی معکوس (\lr{reverse proxy}) متن باز و قدرتمند است که می‌تواند درخواست‌های \lr{HTTP} و \lr{HTTPS} را مدیریت کند، محتوای استاتیک را سریع سرو کند و بار سرور را بین چند اپلیکیشن یا سرویس پخش کند. به دلیل عملکرد بالا و مصرف کم منابع، \lr{Nginx} در پروژه‌های بزرگ وب بسیار محبوب است و در این پروژه مسئول سرو کردن فایل‌های استاتیک شامل \lr{CSS}، \lr{JavaScript}، تصاویر، فونت‌ها و فایل‌های آپلود شده توسط کاربران است که بار پردازشی قابل توجهی را از سرور اپلیکیشن \lr{Django} برمی‌دارد.
	
	همچنین مدیریت گواهی‌های \lr{SSL/TLS} و رمزگذاری ترافیک \lr{HTTPS} بر عهده \lr{Nginx} بوده که شامل تنظیم گواهی، مدیریت سوئیت‌های رمز و اعمال سرآیندهای امنیتی می‌باشد. در صورت استفاده از چندین نمونه از اپلیکیشن، \lr{Nginx} قابلیت توزیع بار بین آنها را فراهم می‌کند تا در مواقع ترافیک بالا عملکرد بهینه حفظ شود.
	
	\subsection{\lr{Daphne}}
	\lr{Daphne} یک سرور \lr{ASGI (Asynchronous Server Gateway Interface)} برای اجرای اپلیکیشن‌های \lr{Django} است که قابلیت پردازش درخواست‌های \lr{HTTP}، \lr{WebSocket} و سایر پروتکل‌های ناهمزمان را دارد. برخلاف سرورهای \lr{WSGI} سنتی، \lr{Daphne} امکان مدیریت اتصالات طولانی‌مدت و \lr{real-time} را فراهم می‌کند که برای سیستم چت و نوتیفیکیشن‌های آنی ضروری است.
	
	\lr{Daphne} درخواست‌های \lr{HTTP} را از \lr{Nginx} دریافت کرده و آن‌ها را به اپلیکیشن \lr{Django} می‌رساند، همچنین اتصالات \lr{WebSocket} را برای ارتباطات لحظه‌ای مدیریت می‌کند. سیستم با قابلیت‌های پیشرفته‌ای شامل مدیریت اتصالات همزمان بالا، پشتیبانی کامل از \lr{Django Channels} و لایه‌های کانال، ادغام با \lr{Redis} برای واسطه‌گری پیام و قابلیت مقیاس‌پذیری افقی برای مدیریت ترافیک بالا تنظیم شده است.
	
	\lr{Daphne} همچنین مکانیزم‌های مدیریت خطا، زمان‌اشام اتصال و خاموشی نرم را پیاده‌سازی کرده تا در محیط تولید پایداری و قابلیت اطمینان مناسبی داشته باشد. این سرور به صورت یکپارچه با سیستم احراز هویت \lr{Django} کار می‌کند و امکان تشخیص کاربر در اتصالات \lr{WebSocket}، مجوز بر اساس نقش کاربر و مدیریت نشست در ارتباطات \lr{real-time} را فراهم می‌آورد.
	
	\section{کنترل نسخه (\lr{Git})}
	\lr{Git} یک سیستم کنترل نسخه (\lr{Version Control System}) متن باز است که برای مدیریت تغییرات در کدها و پروژه‌ها استفاده می‌شود و نقش حیاتی در توسعه و نگهداری سیستم مدیریت خوابگاه ایفا می‌کند. با \lr{Git} می‌توان تغییرات فایل‌ها را پیگیری کرد، به نسخه‌های قبلی بازگشت، شاخه‌های مختلف برای توسعه همزمان ایجاد کرد و با تیم‌های بزرگ به صورت امن همکاری نمود که این امکانات برای پروژه‌ای با پیچیدگی بالا و چندین نوع کاربر مختلف بسیار ضروری است.
	
	\lr{Git} پایه و اساس بسیاری از سرویس‌های میزبانی کد مانند \lr{GitHub} و \lr{GitLab} است و در این پروژه از آن برای مدیریت تمامی فایل‌های سورس کد شامل مدل‌های \lr{Django}، فایل‌های قالب، استایل‌های \lr{CSS}، اسکریپت‌های \lr{JavaScript} و تنظیمات استقرار استفاده شده است.
	
	سیستم شاخه‌بندی پروژه شامل شاخه اصلی برای کد آماده تولید، شاخه توسعه برای ویژگی‌های در حال توسعه، شاخه‌های ویژگی برای هر قابلیت جدید و شاخه‌های اصلاح فوری برای رفع سریع باگ‌های حیاتی طراحی شده که این ساختار امکان توسعه همزمان چندین ویژگی، تست کامل قبل از ادغام، بازگشت آسان در صورت بروز مشکل و حفظ تاریخچه کامل تغییرات را فراهم می‌آورد.
	
	همچنین سیستم برچسب‌گذاری برای مشخص کردن نسخه‌های مختلف سیستم، قراردادهای پیام کامیت برای مستندسازی بهتر تغییرات و قلاب‌های گیت برای اجرای خودکار تست‌ها و بررسی کیفیت کد پیاده‌سازی شده است.
	
	
	\chapter{طریقه کار با سیستم}
	\vspace{-2.5\baselineskip}
	\noindent\rule{\textwidth}{0.4pt}
	
	\hspace{-0.1cm}
	سیستم مدیریت درخواست‌های خوابگاه دانشجویی دارای سه نوع کاربر اصلی است که هرکدام وظایف و دسترسی‌های مختصی دارند. کاربران دانشجو که اصلی‌ترین بهره‌برداران سیستم محسوب می‌شوند و مسئول ثبت درخواست‌ها، پیگیری وضعیت آن‌ها و ارائه بازخورد پس از تکمیل کار هستند.
	
	کاربران مدیر سیستم که نقش کلیدی در بررسی، تأیید یا رد درخواست‌ها، مدیریت اولویت‌بندی و نظارت بر کل فرآیند بر عهده دارند.
	
	همچنین کاربران تیم خدمات فنی که مسئول دریافت درخواست‌های تأیید شده، انجام اقدامات لازم، مستندسازی فرآیند تعمیر از طریق ارسال تصاویر و گزارش نهایی تکمیل کار می‌باشند.
	
	هریک از این گروه‌های کاربری دارای رابط کاربری مختص خود بوده و امکانات و عملکردهای متفاوتی در اختیار دارند که متناسب با نقش و مسئولیت‌هایشان در فرآیند کلی سیستم طراحی شده است. در ادامه این مستند، به تفصیل نحوه کار با سیستم از دیدگاه هریک از این سه نوع کاربر پرداخته خواهد شد تا راهنمای کاملی برای استفاده بهینه از تمامی قابلیت‌های سیستم فراهم شود.
	
	\section{دانشجویان}
	\noindent\rule{\textwidth}{0.4pt}
	
	دانشجویان به عنوان اصلی‌ترین کاربران سیستم، دسترسی به مجموعه کاملی از امکانات و عملکردها دارند که در قالب پورتال اختصاصی دانشجویی ارائه شده است. دانشجویان پس از ورود به سیستم می‌توانند اقدام به ثبت نام اولیه خود در سیستم خوابگاه نموده و اطلاعات شخصی، تماس و تحصیلی خود را تکمیل کنند. پس از تأیید ثبت نام توسط مدیر سیستم، دانشجویان قابلیت ثبت درخواست‌های مختلف تعمیرات و خدمات را خواهند داشت که شامل مسائل برقی، آب و فاضلاب، تهویه مطبوع، نجاری و سایر خدمات عمومی می‌باشد.
	
	هر درخواست ثبت شده توسط دانشجو شامل عنوان مشکل، توضیحات تفصیلی، نوع خدمت مورد نیاز و در صورت لزوم ضمیمه تصاویر مربوط به مشکل خواهد بود. دانشجویان پس از ثبت درخواست، امکان پیگیری وضعیت آن را در مراحل مختلف شامل در انتظار بررسی، تأیید شده، در حال انجام و تکمیل شده دارند. همچنین سیستم امکان دریافت اطلاع‌رسانی‌های خودکار در زمان تغییر وضعیت درخواست‌ها را فراهم آورده است. پس از تکمیل کار توسط تیم خدمات، دانشجویان می‌توانند نسبت به کیفیت خدمات دریافتی امتیاز داده و نظرات و پیشنهادات خود را در سیستم ثبت نمایند که این بازخوردها در بهبود کیفیت خدمات آینده مورد استفاده قرار می‌گیرد.

	\textbf{نکات اضافی:}
	
	علاوه بر فرایند ثبت نام و ورود برای تغییر رمز عبور می‌توان در صفحه لاگین روی \lr{forget password} کلیک کرد و با وارد کردن آدرس ایمیل خود که با آن ثبت نام کرده‌اید، رمز جدید به صورت خودکار برای شما ارسال می‌شود. (برای جلوگیری از حملات \lr{brute force} در زمان وارد کردن آدرس ایمیل خود به شما خطایی نمایش داده نمی‌شود ولی در صورت صحیح بودن، رمز جدید برای شما ارسال می‌شود).
	
	برای ماندن در سیستم بعد از بستن مرورگر می‌توانید تیک گزینه \lr{remember me} را علامت بزنید.

(همچنین در این صفحه گزینه \lr{logout} قرار گرفته شده است).

		\item \textbf{توضیحات تفصیلی}: باید حداقل شامل ۱۰ کاراکتر بوده و اطلاعات کاملی درباره مشکل، موقعیت دقیق آن در اتاق و جزئیات مربوطه ارائه دهد
		\item \textbf{انتخاب محل وقوع مشکل}: از لیست اتاق‌های موجود الزامی بوده که دانشجویان می‌توانند از بین تمامی ساختمان‌ها، طبقات و اتاق‌های مختلف انتخاب نمایند
		\item \textbf{ضمیمه تصویر}: کاملاً الزامی بوده و فایل باید کمتر از ۵ مگابایت و در فرمت‌های معتبر تصویر شامل \lr{JPG}، \lr{PNG}، \lr{GIF} باشد
	\end{itemize}

	بخش تایم‌لاین (\lr{status timeline}) وضعیت که در سمت راست صفحه قرار دارد، تمامی مراحل طی شده توسط درخواست را به ترتیب زمانی نمایش می‌دهد که شامل مراحل ثبت اولیه، تأیید توسط مدیر، اختصاص به متخصص، شروع کار و تکمیل نهایی با ذکر تاریخ و ساعت دقیق هر مرحله می‌باشد.
	
	پس از اختصاص درخواست به متخصص، اطلاعات کامل وی شامل نام، تخصص، امتیاز کلی و تاریخ عضویت در سیستم نمایش داده شده که دانشجویان می‌توانند از سابقه کاری و میزان رضایت‌مندی سایر کاربران نسبت به خدمات ارائه شده توسط آن فرد انجام‌دهنده کار اطلاع یابند. 
	
	پس از تکمیل کار، بخش جزئیات تکمیل شامل یادداشت‌های نهایی متخصص و تصاویر مربوط به وضعیت پس از انجام تعمیرات نمایش داده شده و در نهایت دانشجو امکان امتیازدهی به کیفیت خدمات از یک تا پنج ستاره و ثبت نظرات تکمیلی خود را در قسمت ارزیابی خدمات خواهد داشت که این بازخوردها در بهبود کیفیت خدمات آینده و ارزیابی عملکرد متخصصان مورد استفاده قرار می‌گیرد.

	\subsubsection*{وضعیت‌های مختلف درخواست تعمیرات}
	
	سیستم دارای پنج وضعیت اصلی برای پیگیری مراحل مختلف درخواست‌های تعمیرات می‌باشد:
	
	\begin{itemize}
		\item \textbf{Submitted (ارسال شده)}: نشان‌دهنده ثبت موفق درخواست و انتظار بررسی توسط مدیر
		\item \textbf{Approved (تأیید شده)}: نمایانگر تأیید درخواست و آمادگی برای واگذاری به تیم فنی
		\item \textbf{Assigned to Expert (اختصاص به متخصص)}: به معنای واگذاری درخواست به متخصص مناسب
		\item \textbf{In Progress (در حال انجام)}: نشان‌دهنده شروع فعال کار توسط متخصص
		\item \textbf{Completed (تکمیل شده)}: به معنای پایان موفقیت‌آمیز کار و آمادگی برای دریافت بازخورد
	\end{itemize}
	
	در تمامی این مراحل به صورت خودکار ایمیل اطلاع‌رسانی برای دانشجو ارسال می‌شود که در مرحله تکمیل کار علاوه بر اطلاع‌رسانی، لینک مستقیمی برای امتیازدهی و ثبت نظرات نسبت به کیفیت خدمات دریافتی نیز ارسال می‌شود تا دانشجو بتواند بدون نیاز به ورود مجدد به سیستم، تجربه خود را به اشتراک بگذارد.
	
	\subsection{فرآیند ارزیابی و امتیازدهی خدمات}
	
	دانشجویان پس از تکمیل کار تعمیرات و رسیدن درخواست به وضعیت \textbf{Completed} دو مسیر برای ارزیابی کیفیت خدمات در اختیار دارند:
	
	\subsubsection*{مسیرهای دسترسی به ارزیابی}
	\begin{itemize}
		\item \textbf{لینک مستقیم}: استفاده از لینک مستقیم ارسال شده در ایمیل اطلاع‌رسانی تکمیل کار که بدون نیاز به ورود مجدد به سیستم، دانشجو را مستقیماً به صفحه امتیازدهی هدایت می‌کند
		\item \textbf{داشبورد شخصی}: ورود به داشبورد شخصی، مشاهده درخواست مورد نظر و کلیک بر روی دکمه \textbf{Rate Service} در بخش اقدامات صفحه جزئیات درخواست
	\end{itemize}
	
	در صفحه ارزیابی، خلاصه‌ای از درخواست تکمیل شده شامل عنوان، متخصص مسئول و تاریخ تکمیل نمایش داده شده تا دانشجو بتواند به راحتی درخواست مورد نظر را شناسایی نماید.
	
	\subsubsection*{سیستم امتیازدهی}
	
	بخش اصلی ارزیابی شامل انتخاب امتیاز از یک تا پنج ستاره است که هر امتیاز دارای برچسب توضیحی مناسب می‌باشد:
	
	\begin{itemize}
		\item \textbf{۱ - Poor}: برای خدمات ضعیف
		\item \textbf{۲ - Fair}: برای خدمات متوسط  
		\item \textbf{۳ - Good}: برای خدمات خوب
		\item \textbf{۴ - Very Good}: برای خدمات بسیار خوب
		\item \textbf{۵ - Excellent}: برای خدمات عالی
	\end{itemize}
	
	انتخاب امتیاز کاملاً الزامی بوده و بدون آن امکان ثبت نظر وجود ندارد، در حالی که بخش نظرات و پیشنهادات به صورت اختیاری در نظر گرفته شده و دانشجویان می‌توانند تجربه خود، نکات مثبت، نواقص احتمالی و پیشنهادات بهبود را در این قسمت به اشتراک بگذارند.

		\item امکان استفاده از کلیدهای میانبر شامل اعداد ۱ تا ۵ برای انتخاب سریع امتیاز
		\item ترکیب \lr{Ctrl + Enter} برای ثبت سریع نظر جهت ارائه تجربه کاربری روان و کارآمد

	\subsection{سیستم تیکت‌های پشتیبانی و ارتباط با مدیریت}
	
	دانشجویان علاوه بر امکان ثبت درخواست‌های تعمیرات، امکان ارتباط مستقیم با مدیریت سیستم از طریق سیستم تیکت‌های پشتیبانی را نیز در اختیار دارند که با کلیک بر روی گزینه \textbf{My Tickets} در داشبورد اصلی، وارد صفحه مدیریت تیکت‌ها می‌شوند. این بخش که با عنوان \textbf{"گفتگوی لحظه‌ای با مدیریت"} معرفی شده، امکان تعامل دوطرفه و آنی بین دانشجو و تیم مدیریت را فراهم آورده و برای طرح سوالات عمومی، شکایات، پیشنهادات و مسائل غیرفنی طراحی شده است.
	
	\subsubsection*{داشبورد مدیریت تیکت‌ها}
	
	صفحه تیکت‌ها دارای چهار کارت آماری رنگی است که وضعیت کلی تیکت‌های دانشجو را نمایش می‌دهد:
	
	\begin{itemize}
		\item \textbf{TOTAL TICKETS} با رنگ آبی: نشان‌دهنده کل تیکت‌های ثبت شده در طول زمان
		\item \textbf{PENDING} با رنگ زرد: نمایانگر تیکت‌های در انتظار پاسخ مدیریت
		\item \textbf{ANSWERED} با رنگ سبز: نشان‌دهنده تیکت‌هایی که مدیریت به آن‌ها پاسخ داده
		\item \textbf{CLOSED} با رنگ خاکستری: نمایانگر تیکت‌های حل شده و بسته شده

	\section{متخصصان}
	\noindent\rule{\textwidth}{0.4pt}
	
	متخصصان خدمات فنی یا Service Expert ها کاربران تخصصی سیستم محسوب می‌شوند که مسئول دریافت، بررسی و انجام درخواست‌های تعمیرات تأیید شده توسط مدیر سیستم می‌باشند. پس از ورود به پورتال اختصاصی خود، متخصصان می‌توانند لیست درخواست‌های واگذار شده به خود را مشاهده کرده و وضعیت هر درخواست را به "در حال انجام(In Progress )" تغییر دهند تا شروع فعال کار را اعلام نمایند و در طول فرآیند تعمیرات، آنها موظف به تهیه و آپلود تصاویر قبل و بعد از انجام کار هستند تا مستندات کاملی از فرآیند تعمیرات ارائه دهند و همچنین امکان ثبت یادداشت‌های فنی، گزارش مراحل کاری و اطلاعات تکمیلی مربوط به تعمیرات انجام شده را در سیستم دارند. پس از تکمیل کار، متخصصان باید وضعیت درخواست را به "تکمیل شده(Completed)" تغییر داده و گزارش نهایی خود را ثبت کنند تا فرآیند به مرحله ارزیابی توسط دانشجو برسد و عملکرد آنها بر اساس امتیازات و بازخوردهای دریافتی از دانشجویان ارزیابی و در پروفایل شخصی آنها ثبت می‌شود. متخصصان همچنین امکان ثبت تیکت برای ارتباط مستقیم با مدیریت سیستم را نیز در اختیار دارند تا بتوانند مسائل فنی، درخواست ابزار یا تجهیزات، گزارش مشکلات اجرایی یا ارائه پیشنهادات بهبود خدمات را به صورت منظم و قابل پیگیری با تیم مدیریت در میان بگذارند.
	
	\subsection{نحوه ورود متخصصان خدمات فنی به سیستم}
	
	متخصصان خدمات فنی یا Service Experts ها برای ورود به سیستم ابتدا باید توسط مدیر سیستم به عنوان کاربر جدید ثبت شده و اطلاعات ورود شامل نام کاربری، رمز عبور اولیه و مشخصات تخصص خود را از ادمین دریافت نمایند که این فرآیند کاملاً دستی بوده و امکان خودثبت‌نامی برای متخصصان وجود ندارد. پس از دریافت اطلاعات کاربری از مدیر، متخصصان می‌توانند از طریق لینک مختص Service Expert یا از طریق منوی کشویی صفحه اصلی سیستم، وارد بخش ورود متخصصان شده و با استفاده از نام کاربری و رمز عبور اختصاصی خود به داشبورد تخصصی دسترسی یابند که پس از ورود موفقیت‌آمیز، متخصصان به صفحه مدیریت درخواست‌های اختصاص یافته و ابزارهای کاری مورد نیاز خود هدایت می‌شوند.
	
	\textbf{نمونه شناسه و رمز عبور برای یک متخصص الکترونیک برای ورود و کار با سیستم:}
	\begin{itemize}
		\item \textbf{شناسه کارمندی (\lr{Employee ID})}: \lr{EMP001}
		\item \textbf{رمز عبور (\lr{Password})}: \lr{expert123}
	\end{itemize}
	
	همچنین سایر قابلیت‌ها مانند فراموشی رمز عبور (\lr{forget password}) و ماندن در سیستم (\lr{keep me sign in}) نیز برای متخصصان طراحی شده که تنها تفاوت آن این است که در فراموشی رمز عبور برخلاف دانشجویان که می‌بایست ایمیل خود را وارد کنند، آنها موظف هستند شناسه کارمندی (\lr{Employee ID}) خود را وارد کنند.

	\textbf{\lr{Sites}:} بخش \lr{Sites} مربوط به فریمورک \lr{Sites} جنگو است که اطلاعات اساسی وب‌سایت‌ها را مدیریت می‌کند. این بخش شامل نام دامنه و نام نمایشی هر سایت است. در حال حاضر فقط برای یک خوابگاه طراحی شده ولی پایه‌ای برای گسترش‌های آینده محسوب می‌شود. \lr{Posts} در حالت فعلی در سیستم وجود ندارد ولی فضای آن آماده شده.

	\textbf{\lr{Ticket} و \lr{Ticket message}:} بخش \lr{Ticket} سیستم ارتباط مستقیم و \lr{real-time} بین کاربران و مدیریت را فراهم می‌آورد. در حال حاضر دانشجویان و متخصصان می‌توانند برای مسائل غیرفنی، شکایات، پیشنهادات و سوالات عمومی با مدیریت در تماس باشند. این سیستم دارای وضعیت‌های مختلف شامل در انتظار پاسخ، پاسخ داده شده و بسته شده است.
	
	گسترش‌های آینده شامل سیستم اولویت‌بندی تیکت‌ها بر اساس فوریت و اهمیت، تخصیص خودکار تیکت‌ها به ادمین‌های مختلف بر اساس موضوع، سیستم دسته‌بندی موضوعات برای مدیریت بهتر پیام‌ها و امکان ضمیمه فایل و تصویر در تیکت‌ها خواهد بود. سیستم گزارش‌گیری از عملکرد پاسخگویی، زمان میانگین پاسخ و میزان رضایت کاربران نیز قابل اضافه است. همچنین امکان ایجاد تیکت‌های گروهی برای اطلاع‌رسانی‌های عمومی و سیستم پیگیری خودکار تیکت‌های حل نشده برای جلوگیری از فراموشی پیام‌ها در گسترش‌های بعدی پیاده‌سازی خواهد شد.

	این بخش مرکز اصلی مدیریت درخواست‌های تعمیرات است که تمامی درخواست‌های ثبت شده توسط دانشجویان را نمایش می‌دهد. مدیر از این قسمت می‌تواند درخواست‌ها را بر اساس وضعیت، اولویت، نوع خدمات، تاریخ ثبت، تخصص مورد نیاز، ساختمان و امتیاز دانشجو فیلتر کند. هر درخواست شامل اطلاعات کاملی از جمله عنوان مشکل، نام دانشجو، نوع خدمات، سطح اولویت، وضعیت فعلی، متخصص اختصاص یافته، امتیاز دریافتی از دانشجو و روزهای گذشته از ثبت درخواست است.
	
	سیستم دارای \textbf{شش دسته اقدام گروهی اصلی} است که قابلیت مدیریت حجم بالای درخواست‌های تعمیرات را فراهم می‌آورد. 
	
	\textbf{اقدام تأیید دسته‌ای} (\faCheck\  \lr{Approve selected requests}) که پس از انتخاب درخواست‌های مورد نظر، وضعیت آن‌ها را از \lr{pending} به \lr{approved} تغییر داده و نام کاربری مدیر به همراه زمان دقیق تأیید در فیلدهای \lr{approved\_by\_name} و \lr{approved\_at} ثبت می‌گردد (برای دیدن این فیلدها با کلیک بر روی \lr{ID} می‌توانید مشخصات درخواست را ببینید).
	
	\textbf{اقدام رد دسته‌ای} (\faTimes\  \lr{Reject selected requests}) که درخواست‌های انتخاب شده با وضعیت \lr{pending} را به \lr{rejected} تغییر داده و پیام پیش‌فرض \textit{"Rejected by admin - contact administration for details"} را در فیلد \lr{rejection\_reason} ثبت می‌کند.
	
	\textbf{اقدام اختصاص خودکار متخصص} (\faUser\  \lr{Auto-assign experts}) که بر اساس نوع خدمات مورد نیاز (\lr{service\_type}) در هر درخواست، به صورت هوشمند متخصص مناسب با تخصص مرتبط و وضعیت فعال را پیدا کرده و درخواست را به وی اختصاص می‌دهد. در صورت یافتن متخصص مناسب، فیلدهای \lr{assigned\_expert} و \lr{assigned\_at} به‌روزرسانی شده و چنانچه درخواست در وضعیت \lr{pending} باشد، به طور خودکار به \lr{approved} تغییر می‌یابد.
	
	\textbf{اقدامات تنظیم اولویت} شامل سه سطح می‌باشد که \textbf{تنظیم اولویت بالا} (\textcolor{red}{\faCircle}\  \lr{Set high priority}) برای مسائل فوری و حساس، \textbf{تنظیم اولویت متوسط} (\textcolor{orange}{\faCircle}\  \lr{Set medium priority}) برای مسائل عادی و \textbf{تنظیم اولویت پایین} (\textcolor{green}{\faCircle}\  \lr{Set low priority}) برای مسائل غیر ضروری طراحی شده است. این اقدامات فیلد \lr{priority} درخواست‌های انتخاب شده را به ترتیب به \lr{high}، \lr{medium} و \lr{low} تنظیم نموده تا متخصصان بتوانند کارهای خود را بر اساس فوریت اولویت‌بندی کنند.
	
	\textbf{اقدام تنظیم در حال انجام} (\faSync\  \lr{Mark in progress}) که درخواست‌های تأیید شده با وضعیت \lr{approved} را به \lr{in\_progress} تغییر داده و فیلد \lr{work\_started\_at} را روی زمان فعلی تنظیم می‌کند تا نشان دهد کار عملاً آغاز شده است.
	
	\textbf{اقدام تنظیم تکمیل شده} (\faCheck\  \lr{Mark completed}) که درخواست‌های در حال انجام با وضعیت \lr{in\_progress} را به \lr{completed} تغییر داده و فیلد \lr{completed\_at} را به‌روزرسانی می‌کند تا نشان دهد کار به پایان رسیده و آماده دریافت بازخورد از دانشجو است.
	
	\subsubsection*{قانون کنترل اولویت‌بندی}
	
	\textbf{قانون اساسی سیستم} این است که هیچ درخواستی با وضعیت \lr{priority} برابر با \lr{not\_decided} نمی‌تواند تأیید شود. در صورت تلاش مدیر برای تأیید درخواست‌هایی که اولویت آن‌ها تعیین نشده، \textbf{سیستم هشدار خطای مفصل} نمایش می‌دهد که شامل پیام اصلی \lr{"Cannot approve requests without priority"} به همراه \textbf{لیست دقیق عناوین درخواست‌های مشکل‌دار} می‌باشد. چنانچه تعداد درخواست‌های بدون اولویت بیش از سه مورد باشد، سیستم سه عنوان اول را نمایش داده و عبارت \textit{"and X more"} را اضافه می‌کند تا حجم پیام خطا کنترل شود.

	\subsection{مدیریت متخصصان خدمات فنی (\lr{Service Experts})}
	
	این بخش مختص مدیریت متخصصان خدمات فنی است که تمامی کارکنان فنی سیستم را نمایش می‌دهد. هر متخصص دارای مشخصات کاملی شامل نام کاربری، کد کارمندی، تخصص، وضعیت فعالیت، بار کاری فعلی، تعداد کارهای تکمیل شده و امتیاز میانگین از دانشجویان است.
	
	هر متخصص دارای \textbf{هفت حوزه تخصصی} اصلی است که شامل \textbf{خدمات برقی، لوله‌کشی و آب و فاضلاب، سیستم‌های تهویه مطبوع، نجاری و کارهای چوبی، تعمیرات عمومی، نظافت و خدمات بهداشتی و سیستم‌های امنیتی} می‌باشد. سیستم به گونه‌ای طراحی شده که درخواست‌های تعمیرات تنها به متخصصانی با تخصص مرتبط نمایش داده شوند تا بهره‌وری و کیفیت کار افزایش یابد. علاوه بر تخصص، هر متخصص دارای کد کارمندی منحصربه‌فرد، امتیاز میانگین بر اساس ارزیابی دانشجویان و وضعیت فعالیت قابل کنترل توسط مدیر است.
	
	سیستم دارای \textbf{شش دسته اقدام گروهی} برای مدیریت کارآمد متخصصان خدمات فنی طراحی شده که امکان کنترل دقیق عملکرد و وضعیت کاری تیم فنی را فراهم می‌آورد.
	
	\textbf{اقدام فعال‌سازی متخصصان} (\faCheck\  \lr{Activate experts}) که حساب‌های کاربری غیرفعال را مجدداً به وضعیت فعال تبدیل کرده و فیلد \lr{is\_active} را روی \lr{True} تنظیم می‌کند تا متخصص بتواند مجدداً وارد سیستم شده و درخواست‌های جدید را دریافت نماید. 
	
	\textbf{اقدام غیرفعال‌سازی} (\faBan\  \lr{Deactivate experts}) که متخصصان انتخاب شده را غیرفعال کرده و دسترسی آن‌ها به سیستم قطع می‌گردد، در این حالت متخصص نمی‌تواند وارد سیستم شود و درخواست‌های جدیدی نیز به وی اختصاص نمی‌یابد.
	
	\textbf{اقدام مشاهده بار کاری} (\faChartBar\  \lr{View workload}) که گزارش تفصیلی از وضعیت کاری هر متخصص را در قالب متن ساختاریافته ارائه می‌دهد. این گزارش شامل \textbf{تعداد درخواست‌های فعلی اختصاص یافته} با وضعیت \lr{approved} و \lr{in\_progress}، \textbf{تعداد درخواست‌های تکمیل شده در ماه گذشته} برای سنجش میزان فعالیت اخیر و \textbf{امتیاز میانگین فعلی} بر اساس ارزیابی‌های دریافتی از دانشجویان می‌باشد. نمونه خروجی این گزارش به صورت \textit{"متخصص الف: 3 assigned, 15 completed (last month), 4.2/5 rating"} نمایش داده می‌شود.
	
	\textbf{اقدام مشاهده کارهای تکمیل شده} (\faCheck\  \lr{View completed}) که آمار کلی درخواست‌های به پایان رسیده توسط هر متخصص را همراه با میانگین امتیاز دریافتی نمایش می‌دهد تا مدیر بتواند عملکرد طولانی‌مدت هر متخصص را ارزیابی کرده و در تصمیم‌گیری‌های آتی از این اطلاعات استفاده نماید.
	
	\textbf{اقدام تکمیل اجباری} (\faBolt\  \lr{Force complete}) که برای حل مشکل درخواست‌های معلق در وضعیت \lr{in\_progress} طراحی شده است. این اقدام تمامی درخواست‌هایی که توسط متخصصان انتخاب شده در حال انجام بوده ولی برای مدت طولانی تکمیل نشده‌اند را به صورت اجباری به وضعیت \lr{completed} تبدیل کرده و فیلدهای \lr{completed\_at} را روی زمان فعلی و \lr{completion\_notes} را روی \textit{"Force completed by [نام مدیر]"} تنظیم می‌کند. این اقدام معمولاً در مواقع اضطراری یا هنگام تغییر متخصص استفاده می‌شود.
	
	\textbf{اقدام ریست خودکار رمز عبور} (\faLock\  \lr{Reset expert passwords}) که فرآیند پیچیده‌ای برای تولید و ارسال رمز عبور جدید انجام می‌دهد. این سیستم از \textbf{\lr{random password generator}} استفاده کرده و رمز عبور امن ۸ کاراکتری تولید می‌کند که شامل ترکیب حروف بزرگ و کوچک انگلیسی به همراه اعداد است. پس از تولید رمز، سیستم آن را به صورت \lr{hash} شده در دیتابیس ذخیره نموده و از طریق \textbf{\lr{EmailService}} ایمیلی حاوی رمز جدید، نام کاربری و راهنمای ورود برای متخصص ارسال می‌کند.

	\subsection{دانشجویان (\lr{Student})}
	
	در منوی مربوطه به دانشجویان به مدیریت ساختمان‌های خوابگاه، اتاق‌ها و دانشجویان به صورت کامل پرداخته شده.
	
	\subsubsection*{بخش ساختمان‌ها (\lr{Buildings})}
	
	این بخش مختص مدیریت ساختمان‌های خوابگاه است که مدیر می‌تواند اطلاعات کلی ساختمان‌ها را مشاهده و مدیریت کند. هر ساختمان دارای مشخصات اصلی شامل \textbf{نام ساختمان، تعداد کل طبقات و تاریخ ایجاد} می‌باشد. مدیر از این بخش می‌تواند ساختمان‌های جدید اضافه کرده، ساختمان‌های موجود را ویرایش نماید و بر اساس نام ساختمان، تعداد طبقات و تاریخ ایجاد فیلتر و جستجو انجام دهد.
	
	سیستم قابلیت مرتب‌سازی ساختمان‌ها بر اساس نام و امکان جستجوی سریع در نام‌های ساختمان‌ها را فراهم آورده است. این بخش پایه‌ای برای مدیریت کلی محیط خوابگاه و تخصیص اتاق‌ها محسوب می‌شود که تمامی عملیات مربوط به اتاق‌ها و دانشجویان بر اساس ساختار ساختمان‌های تعریف شده انجام می‌پذیرد. \textbf{(بعد از اضافه کردن ساختمان نیاز به اضافه کردن دستی آن‌ها به لیست ساختمان‌های موجود در هنگام ثبت‌نام دانشجو نیست زیرا این فرآیند به صورت خودکار انجام می‌شود).}

	\subsubsection*{مدیریت دانشجویان (\lr{Students})}
	
	این بخش مرکز اصلی مدیریت دانشجویان ساکن خوابگاه است که تمامی کاربران دانشجویی سیستم را نمایش می‌دهد. هر دانشجو دارای اطلاعات کاملی شامل \textbf{شماره دانشجویی، نام کاربری، آدرس ایمیل، اطلاعات اتاق اختصاصی، وضعیت ثبت‌نام (\lr{pending/approved/rejected})، وضعیت فعالیت حساب و تاریخ ایجاد حساب} می‌باشد.
	
	\textbf{سیستم نمایش اطلاعات دانشجویان:}
	
	صفحه اصلی این بخش جدولی جامع از تمامی دانشجویان نمایش می‌دهد که شامل \textbf{ستون شماره دانشجویی} با قابلیت مرتب‌سازی، \textbf{ستون نام کاربری} که با حساب کاربری اصلی مرتبط است، \textbf{ستون آدرس ایمیل} برای تماس و اطلاع‌رسانی، \textbf{ستون اطلاعات اتاق} که به صورت ساختمان - طبقه - شماره اتاق نمایش داده شده و در صورت عدم تخصیص اتاق \textit{"No room assigned"} نمایش می‌یابد، \textbf{ستون وضعیت فعالیت} که با نشان تیک سبز یا ضربدر قرمز نمایش داده می‌شود و \textbf{ستون تاریخ ایجاد} که زمان دقیق ثبت‌نام اولیه را نشان می‌دهد.
	
	\textbf{سیستم فیلترینگ و جستجوی پیشرفته:}
	
	سیستم دارای \textbf{چهار فیلتر اصلی} برای دسته‌بندی دانشجویان است که شامل \textbf{فیلتر وضعیت ثبت‌نام} با گزینه‌های (\lr{pending/approved/rejected})، \textbf{فیلتر ساختمان اقامت} برای نمایش دانشجویان ساکن در ساختمان مشخص، \textbf{فیلتر طبقه} برای جستجوی دقیق‌تر در سطح طبقات و \textbf{فیلتر تاریخ ایجاد} برای یافتن دانشجویان ثبت‌نام شده در بازه زمانی مشخص می‌باشد. علاوه بر فیلترها، \textbf{نوار جستجوی هوشمند} امکان جستجو در سه فیلد اصلی شامل شماره دانشجویی، نام کاربری و آدرس ایمیل را به صورت همزمان فراهم آورده تا مدیر بتواند با تایپ بخشی از اطلاعات، دانشجوی مورد نظر را سریعاً پیدا کند.
	
	سیستم دارای \textbf{پنج اقدام گروهی اصلی} است که قابلیت انجام عملیات روی چندین دانشجو به صورت هم
	زمان را فراهم می‌آورد. \textbf{اقدام تأیید دسته‌ای} (\faCheck\  \lr{Approve selected students}) که پس از انتخاب دانشجویان با وضعیت \lr{pending} یا \lr{rejected} آن‌ها را به صورت خودکار به وضعیت \lr{approved} تغییر داده و نام کاربری مدیر به همراه زمان دقیق تأیید در سیستم ثبت می‌شود. \textbf{اقدام رد دسته‌ای} (\faTimes\  \lr{Reject selected students}) که دانشجویان انتخاب شده با وضعیت \lr{pending} را به \lr{rejected} تغییر داده و دلیل پیش‌فرض \textit{"Rejected by admin"} را ثبت می‌کند.
	
	\textbf{اقدام فعال‌سازی حساب‌ها} (\faCheck\  \lr{Activate selected students}) که حساب‌های کاربری غیرفعال را مجدداً فعال نموده و دانشجو قابلیت ورود به سیستم را بازیابد، \textbf{اقدام غیرفعال‌سازی} (\faBan\  \lr{Deactivate selected students}) که حساب‌های مشکوک یا متخلف را غیرفعال کرده و دسترسی آن‌ها به سیستم قطع می‌گردد، و \textbf{اقدام ریست رمز عبور} (\faLock\  \lr{Reset student passwords}) که رمز عبور امن ۸ کاراکتری جدید تولید کرده و از طریق ایمیل برای دانشجو ارسال می‌نماید (برای دیدن این پنل می‌توانید در نوار سمت چپ بر روی \lr{student} کلیک کنید).
	
	\textbf{فرآیند تأیید و رد درخواست‌های ثبت‌نام:}
	
	هنگامی که مدیر اقدام به تأیید یا رد دانشجویان می‌کند، سیستم \textbf{فرآیند خودکار ارسال ایمیل} را آغاز می‌نماید. در صورت تأیید، ایمیلی حاوی اطلاعات ورود شامل نام کاربری و دستورالعمل‌های اولیه استفاده از سیستم برای دانشجو ارسال شده و در صورت رد، ایمیلی حاوی دلیل رد درخواست و راهنمایی‌های لازم برای رفع مشکل ارسال می‌گردد.
	
	\subsection{سیستم اطلاع‌رسانی هوشمند (\lr{Notification System})}
	
	این بخش یکی از مهم‌ترین ابزارهای مدیریتی سیستم است که به مدیر امکان \textbf{پیگیری لحظه‌ای تمامی فعالیت‌های جدید} را می‌دهد تا هیچ موردی از دست نرود و بتواند با اولویت صحیح به مسائل مختلف رسیدگی نماید.
	
	\textbf{نوار اطلاع‌رسانی بالای صفحه:} در \textbf{گوشه بالا راست} تمامی صفحات مدیریت، نوار اطلاع‌رسانی قرار دارد که هر \textbf{۱۰ ثانیه به صورت خودکار} وضعیت جدیدترین موارد را بررسی می‌کند. در صورت وجود موارد جدید، \textbf{دکمه زرد رنگ} با نماد زنگ و عدد کل موارد جدید نمایش داده می‌شود و در صورت عدم وجود مورد جدید، \textbf{دکمه خاکستری} بدون عدد نمایش می‌یابد.
	
	\textbf{انواع موارد قابل پیگیری:} سیستم \textbf{چهار نوع فعالیت اصلی} را پیگیری می‌کند که شامل \textbf{درخواست‌های ثبت‌نام جدید دانشجویان} که منتظر تأیید یا رد توسط مدیر هستند، \textbf{درخواست‌های تعمیرات جدید} که نیاز به بررسی و تصمیم‌گیری دارند، \textbf{تیکت‌های جدید} ایجاد شده توسط دانشجویان و متخصصان برای ارتباط با مدیریت و \textbf{پیام‌های جدید} در تیکت‌هایی که هنوز پاسخ داده نشده‌اند می‌باشد.

	\subsection{داشبورد مدیریت تیکت‌ها و چت}
	
	زمانی که پیام جدیدی از سوی دانشجویان یا متخصصان در تیکت‌ها ارسال می‌شود، \textbf{سیستم نوتیفیکیشن} در بالای داشبورد اصلی ادمین فعال شده و تعداد پیام‌های خوانده نشده را نمایش می‌دهد. ادمین با کلیک بر روی \textbf{آیکون نوتیفیکیشن} و سپس انتخاب گزینه \textbf{\lr{New Messages}} یا \textbf{\lr{New Tickets}} وارد پنل مدیریت تیکت‌ها می‌شود.
	
	\textbf{اگر در سیستم هیچ پیام جدیدی موجود نباشد اما ادمین برای بازبینی دوباره می‌خواهد وارد پنل تیکت شود باید وارد آدرس \lr{https://www.dormitoryplus.ir/ticket/admin-chat/} شود. اما در صورت وجود پیام جدید از طریق نوتیفیکیشن در دسترس است.}
	
	صفحه داشبورد تیکت‌ها شامل \textbf{چهار کارت آماری رنگی} در بالای صفحه است که \textbf{کارت آبی \lr{Total Tickets}} نمایانگر کل تیکت‌های ثبت شده در سیستم، \textbf{کارت زرد \lr{Pending}} نشان‌دهنده تیکت‌هایی که منتظر پاسخ ادمین هستند، \textbf{کارت سبز \lr{Answered}} مربوط به تیکت‌هایی که ادمین به آن‌ها پاسخ داده و \textbf{کارت آبی \lr{Active Chats}} تعداد گفتگوهای فعال را نمایش می‌دهد.

	\subsubsection*{سیستم فیلترینگ و نمایش تیکت‌ها}
	
	داشبورد دارای \textbf{سه فیلتر اصلی} است که شامل دکمه \textbf{\lr{All Tickets}} برای نمایش تمامی تیکت‌ها، \textbf{\lr{Pending}} برای نمایش تیکت‌های در انتظار پاسخ و \textbf{\lr{Answered}} برای تیکت‌هایی که پاسخ داده شده‌اند می‌باشد. جدول اصلی تیکت‌ها شامل ستون‌های \textbf{\lr{ID}} با نمایش شماره تیکت و تعداد پیام‌های خوانده نشده، \textbf{\lr{Title \& User}} شامل عنوان تیکت، نام کاربر و نوع کاربر (\lr{Student/Expert})، \textbf{\lr{Status}} با نشان‌های رنگی مختلف، \textbf{\lr{Messages}} با تعداد کل پیام‌ها و پیام‌های جدید، \textbf{\lr{Last Activity}} نمایانگر آخرین فعالیت و \textbf{\lr{Actions}} شامل دکمه \lr{Chat} برای ورود به گفتگو می‌باشد.
	
	پس از کلیک بر روی دکمه \textbf{\lr{Chat}} در هر تیکت، ادمین وارد محیط گفتگوی زنده می‌شود. در بالای صفحه هدر تیکت قرار دارد که شماره تیکت و موضوع را همراه با وضعیت اتصال نمایش می‌دهد. در همین قسمت سه دکمه مهم وجود دارد: دکمه زرد \textbf{\lr{Close Ticket}} برای بستن تیکت، دکمه \textbf{\lr{Back to Dashboard}} برای بازگشت به لیست و دکمه قرمز \textbf{\lr{Admin Panel}} برای رفتن به پنل اصلی.

	در ادامه اطلاعات کاربر و تیکت در کادر خاکستری نمایش داده می‌شود که شامل نام کاربری همراه با نوع کاربر (دانشجو یا متخصص)، ایمیل کاربر، وضعیت فعلی تیکت (در انتظار پاسخ، پاسخ داده شده یا بسته شده)، تاریخ ایجاد و آخرین فعالیت و متن اصلی مسئله است.
	
	بخش اصلی صفحه مربوط به ناحیه گفتگو است که تمامی پیام‌ها به ترتیب زمانی نمایش داده می‌شود. پیام‌های کاربر در سمت چپ با پس‌زمینه روشن و پیام‌های ادمین در سمت راست با پس‌زمینه قرمز و برچسب \textbf{"\lr{ADMIN}"} قرار می‌گیرند.
	
	در پایین صفحه کادر بزرگ متنی برای نوشتن پاسخ و دکمه \textbf{\lr{Send as Admin}} برای ارسال پیام قرار دارد. سیستم اطلاع می‌دهد که کاربر پس از ارسال پیام، ایمیل اطلاع‌رسانی دریافت می‌کند و می‌توان از کلید ترکیبی \lr{Ctrl+Enter} برای ارسال سریع استفاده کرد.
